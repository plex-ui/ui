export const loadingCode = `const [loading, setLoading] = useState(true)

useEffect(() => {
  if (!loading) return
  const timer = setTimeout(() => setLoading(false), 1500)
  return () => clearTimeout(timer)
}, [loading])

{loading ? (
  <View accessibilityState={{ busy: true }} accessibilityLabel="Loading profile" style={{ flexDirection: "row", alignItems: "center", gap: 12 }}>
    <Skeleton circle height={44} />
    <View style={{ flex: 1, gap: 8 }}>
      <Skeleton height={17} width="50%" />
      <Skeleton height={15} width="80%" />
    </View>
  </View>
) : (
  <View style={{ flexDirection: "row", alignItems: "center", gap: 12 }}>
    <Avatar name="Ada Lovelace" color="info" size={44} />
    <View style={{ flex: 1 }}>
      <Text variant="headline">Ada Lovelace</Text>
      <Text variant="subheadline" color="secondary">Wrote the first program</Text>
    </View>
  </View>
)}
<Button size="sm" variant="soft" disabled={loading} onPress={() => setLoading(true)}>Reload</Button>`

<UsageBlock>{`import { Skeleton } from "@plexui/react-native"`}</UsageBlock>

The props follow `@plexui/ui`'s Skeleton. Both platforms draw it alike from the Plex tokens: a block of `--skeleton-background` with `--skeleton-radius` corners that pulses between full and half opacity every 1.5 seconds, and stands still when the system asks for reduced motion. Skeletons are hidden from screen readers: say what loads on their container with `accessibilityState={{ busy: true }}` and a label.
