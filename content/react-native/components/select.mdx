export const overviewCode = `const FRUITS = [
  { value: "apple", label: "Apple" },
  { value: "banana", label: "Banana", description: "Ripe this week" },
  { value: "cherry", label: "Cherry" },
  { value: "durian", label: "Durian", disabled: true },
]

const [fruit, setFruit] = useState("banana")

<Select aria-label="Fruit" options={FRUITS} value={fruit} onChange={(option) => setFruit(option.value)} />`

export const placeholderCode = `const [fruit, setFruit] = useState(null)

<Select
  aria-label="Fruit"
  options={FRUITS}
  value={fruit}
  onChange={(option) => setFruit(option.value)}
  placeholder="Choose a fruit"
/>`

export const variantsCode = `<Select aria-label="Outline" options={FRUITS} value={outline} onChange={(option) => setOutline(option.value)} />
<Select aria-label="Soft" variant="soft" options={FRUITS} value={soft} onChange={(option) => setSoft(option.value)} />
<Select aria-label="Ghost" variant="ghost" options={FRUITS} value={ghost} onChange={(option) => setGhost(option.value)} />`

export const statesCode = `<Select aria-label="Invalid" options={FRUITS} value={null} onChange={() => {}} placeholder="Required" invalid />
<Select aria-label="Disabled" options={FRUITS} value="apple" onChange={() => {}} disabled />`

export const hugCode = `const SORTS = [
  { value: "recent", label: "Most recent" },
  { value: "popular", label: "Most popular" },
  { value: "price", label: "Lowest price" },
]

const [sort, setSort] = useState("recent")

<Select aria-label="Sort by" options={SORTS} value={sort} onChange={(option) => setSort(option.value)} block={false} pill />`

<UsageBlock>{`import { Select } from "@plexui/react-native"`}</UsageBlock>

The props follow `@plexui/ui`'s Select: `options`, `value` and `onChange`, which receives the chosen option. The field is drawn as an [Input](/react-native/components/input) of the same variant, with the chosen label and an indicator, and a press opens a menu of the options with a checkmark at the chosen one. On iOS that is UIKit's pop-up button: the up and down chevron, and the glass menu of iOS 26 at least 250pt wide. On Android it is Material 3's exposed dropdown menu: the dropdown arrow, which turns up while the menu is open, and a menu as wide as the field, opening at its edge. The menu is a modal layer over the screen; on this page it opens over the page.
