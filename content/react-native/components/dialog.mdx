export const overviewCode = `import { Button, Dialog, Input } from "@plexui/react-native"

const [open, setOpen] = useState(false)
const [actions, setActions] = useState("2") // or "1", "3"
const [destructive, setDestructive] = useState(false) // with two actions
const [preferred, setPreferred] = useState(true) // the main action is filled
const [message, setMessage] = useState(true)
const [field, setField] = useState(false) // with two actions, not destructive
const [stacked, setStacked] = useState(false) // iOS: two actions, one under the other
const [name, setName] = useState("Untitled")

// What the alert says for each kind
function alertText(field, actions, destructive) {
  if (field) return { title: "Rename document", message: "Enter a new name for this document.", action: "Rename" }
  if (actions === "1") return { title: "Unable to save", message: "Check your connection and try again.", action: "OK" }
  if (destructive) return { title: "Delete document?", message: "This document will be deleted. You can’t undo this.", action: "Delete" }
  return { title: "Save changes?", message: "Do you want to save the changes you made to this document?", action: "Save" }
}
const text = alertText(field, actions, destructive)

const main = {
  label: text.action,
  variant: destructive ? "destructive" : "default",
  preferred: !destructive && preferred,
  disabled: field && !name.trim(),
}
const cancel = { label: "Cancel", variant: "cancel" }

<Button color="secondary" variant="outline" onPress={() => setOpen(true)}>
  Open alert
</Button>
<Dialog
  open={open}
  onOpenChange={setOpen}
  title={text.title}
  description={message ? text.message : undefined}
  actions={
    actions === "1" ? [main]
    : actions === "2" ? [cancel, main]
    : [main, { label: "Don’t save", variant: "destructive" }, cancel]
  }
  mobileLayout={stacked ? "stacked" : undefined}
>
  {field && (
    <Input
      accessibilityLabel="Document name"
      variant="soft"
      size="3xl"
      pill // iOS
      value={name}
      onChangeText={setName}
      selectTextOnFocus
    />
  )}
</Dialog>`

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Alert | Dialog |
| Look | Glass panel with capsule actions | A white (night: dark gray) panel with 21dp corners, text buttons in the accent at the end |

iOS calls its sheets dialogs too; for a sheet use [Bottom Sheet](/react-native/components/bottom-sheet).
