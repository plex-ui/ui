export const overviewCode = `import { ContextMenu, List, ListItem } from "@plexui/react-native"
import { Copy, RemoveTrash, Share } from "@plexui/react-native/icons"

<List>
  {notes.map((note) => (
    <ContextMenu key={note.id}>
      <ContextMenu.Trigger>
        <ListItem label={note.title} description={note.detail} onPress={() => open(note)} />
      </ContextMenu.Trigger>
      <ContextMenu.Content aria-label={note.title}>
        <ContextMenu.Item icon={<Copy />} onSelect={() => copy(note)}>Copy</ContextMenu.Item>
        <ContextMenu.Item icon={<Share />} onSelect={() => share(note)}>Share</ContextMenu.Item>
        <ContextMenu.Separator />
        <ContextMenu.Item icon={<RemoveTrash />} danger onSelect={() => remove(note)}>Delete</ContextMenu.Item>
      </ContextMenu.Content>
    </ContextMenu>
  ))}
</List>`

export const ownShapeCode = `import { ContextMenu, Text } from "@plexui/react-native"
import { Pressable } from "react-native"

<ContextMenu>
  <ContextMenu.Trigger radius={18} shape="element">
    <Pressable onLongPress={() => {}} style={{ borderRadius: 18, paddingHorizontal: 14, paddingVertical: 9, backgroundColor: bubble }}>
      <Text>Yes, 7 at the usual place</Text>
    </Pressable>
  </ContextMenu.Trigger>
  <ContextMenu.Content aria-label="Message">
    <ContextMenu.Item icon={<Reply />} onSelect={reply}>Reply</ContextMenu.Item>
    <ContextMenu.Item icon={<Copy />} onSelect={copy}>Copy</ContextMenu.Item>
    <ContextMenu.Separator />
    <ContextMenu.Item icon={<RemoveTrash />} danger onSelect={remove}>Delete</ContextMenu.Item>
  </ContextMenu.Content>
</ContextMenu>`

### Overview

  <p>A touch and hold on the element opens its menu after half a second; a tap still runs the element's own press, here opening the note. On the web a right click opens it too. On iOS 27 the screen dims (black at 20%, 50% at night, over a 30px blur on the web) and the element stays clear over it, lifted, its corners rounded to half its height up to 30 and a soft shadow around it, as iOS lifts a context menu's preview; the glass menu of <a href="/react-native/components/menu">Menu</a> stands 16 under it, or over it when the element is in the lower half of the screen, and keeps to where the element stands: its left edge in the left third of the screen, its right edge in the right third, its centre otherwise (a full-width row, as here). With <code>open</code> the menu measures the element as it opens; a turn of the screen closes it. Delete takes the note off the list. ## Platforms

| | iOS 27 | Android |
| --- | --- | --- |
| Native component | Context menu (UIContextMenuInteraction, `.contextMenu`) | Popup menu on a long press |
| Look | The element lifted over a dimmed, blurred screen, the glass menu 16 from it by the third of the screen it is in | Material's popup under the element at its end, the screen dimmed around it |

### ContextMenu.Trigger

<PropsTable data={[
  { prop: "children", type: "ReactElement", description: "The element: a ListItem, a Pressable, anything with onLongPress (required). Its own onPress and onLongPress still run; the iOS preview is drawn again from its props" },
  { prop: "style", type: "StyleProp<ViewStyle>", description: "Style of the wrapper the element is measured in" },
  { prop: "radius", type: "number", description: "The element's own corner radius: the iOS preview keeps these corners (half the element's height at most), and Android cuts the dimming around the element at them. Unset: the kit's 30 on iOS, square on Android" },
  { prop: "shape", type: '"card" | "element"', default: '"card"', description: "card lifts the element on the preview's card (a grouped surface, clipped); element lifts an element that draws its own shape as it is, with no card and no clip, its shadow following radius" },
]} />
