export const overviewCode = `import { ContextMenu, List, ListItem } from "@plexui/react-native"
import { Copy, RemoveTrash, Share } from "@plexui/react-native/icons"

<List>
  {notes.map((note) => (
    <ContextMenu key={note.id}>
      <ContextMenu.Trigger>
        <ListItem label={note.title} description={note.detail} onPress={() => open(note)} />
      </ContextMenu.Trigger>
      <ContextMenu.Content aria-label={note.title}>
        <ContextMenu.Item icon={<Copy />} onSelect={() => copy(note)}>Copy</ContextMenu.Item>
        <ContextMenu.Item icon={<Share />} onSelect={() => share(note)}>Share</ContextMenu.Item>
        <ContextMenu.Separator />
        <ContextMenu.Item icon={<RemoveTrash />} danger onSelect={() => remove(note)}>Delete</ContextMenu.Item>
      </ContextMenu.Content>
    </ContextMenu>
  ))}
</List>`

<UsageBlock>{`import { ContextMenu } from "@plexui/react-native"`}</UsageBlock>

A menu from a button is a [Menu](/react-native/components/menu).

## Platforms

| | iOS 27 | Android |
| --- | --- | --- |
| Native component | Context menu (UIContextMenuInteraction, `.contextMenu`) | Popup menu on a long press |
| Look | The element lifted over a dimmed, blurred screen, the glass menu 16 from it by the third of the screen it is in | Material's popup under the element at its end, the screen dimmed around it |
