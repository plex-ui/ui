import { RNExample } from "@/components/docs/react-native/RNExample"

export const overviewCode = `import { List, ListItem, Switch, ThemeProvider } from "@plexui/react-native"

// The app's on color, set once on its ThemeProvider: "primary" (the default) or "accent"
const [selectionColor, setSelectionColor] = useState<"primary" | "accent">("accent")
const [description, setDescription] = useState(false)
const [disabled, setDisabled] = useState(false)

<ThemeProvider selectionColor={selectionColor}>
  {/* Each switch is the trailing control of a row, which holds its label and description; a disabled row dims them */}
  <List>
    <ListItem
      label="Notifications"
      description={description ? "Messages and mentions." : undefined}
      disabled={disabled}
      trailing={<Switch accessibilityLabel="Notifications" defaultChecked disabled={disabled} />}
    />
    <ListItem
      label="Weekly summary"
      description={description ? "Sent on Mondays." : undefined}
      disabled={disabled}
      trailing={<Switch accessibilityLabel="Weekly summary" disabled={disabled} />}
    />
  </List>
</ThemeProvider>`

export const controlledCode = `const [checked, setChecked] = useState(false)

<List>
  <ListItem
    label="Airplane mode"
    trailing={<Switch color="accent" accessibilityLabel="Airplane mode" checked={checked} onCheckedChange={setChecked} />}
  />
</List>`

### Overview

<ComponentPreview details={<>
  <p>On a phone, put switches in labelled rows with a trailing control: a <code>ListItem</code> holds the label and description, and the <code>Switch</code> ends the row. The overview and the controlled example compare one row or two; the settings screen shows them in context.</p>
  <p>One setting on and one off, each the <code>trailing</code> control of a <code>ListItem</code>; give the switch the row's label as its <code>accessibilityLabel</code>. On iOS it is the 63x28 track of the iOS 26 switch with a 37x24 capsule thumb; on Android a 31x14dp track under a 20dp thumb, a ring of the track's color around a knob of the card's color, the track turning from gray to the on color as the thumb slides in 200ms. There it stands 22dp in from the row's end and draws no press of its own: the row shows the press.</p>
  <p>Without <code>color</code> a switch takes the <code>selectionColor</code> of its <code>ThemeProvider</code>, set once for the app: <code>"primary"</code> (the default) is the Plex primary, the ink, as on the web (on iOS at night the switch turns the web's blue, <code>--switch-track-color-checked</code>), and <code>"accent"</code> is each platform&apos;s own on color, the system green on iOS and the accent on Android. <code>color</code> on one switch overrides it. A row's <code>description</code> puts a second line under its label. A switch with a <code>label</code> of its own takes <code>labelPosition</code>; in a row the label stays at the start and the switch trails it, so the example does not offer it. <code>disabled</code> keeps a switch as it is: on retains a muted color, off the neutral track (on Android the switch fades to half), and it takes no presses; pass it to the row as well, so its label and description dim with it.</p>
  <p>Use <code>defaultChecked</code> for an initial value.</p>
</>} code={overviewCode} previewMinHeight={168}>

</ComponentPreview>
