export const overviewCode = `import { Button, Menu, Toolbar } from "@plexui/react-native"
import { CheckCircle, Desktop, DotsHorizontal, FileDocument, FolderPlus, List, VideoGrid } from "@plexui/react-native/icons"

const [icons] = useState(true)
const [checkmarks] = useState(true)
const [view, setView] = useState("list")
const [sort, setSort] = useState("date")
const [group, setGroup] = useState("date")
const SORTS = [["name", "Name"], ["kind", "Kind"], ["date", "Date"], ["size", "Size"], ["tags", "Tags"]]
const GROUPS = [["none", "None"], ["kind", "Kind"], ["date", "Date"], ["size", "Size"], ["shared", "Shared by"]]

// The More button of the top bar opens the menu
<Toolbar titleStyle="large">
  <Toolbar.Title>Overview</Toolbar.Title>
  <Toolbar.Trailing>
    <Menu>
      <Menu.Trigger>
        <Button color="secondary" variant="soft" uniform accessibilityLabel="More"><DotsHorizontal /></Button>
      </Menu.Trigger>
      <Menu.Content align="end" aria-label="Downloads">
        <Menu.Item icon={icons ? <CheckCircle /> : undefined} onSelect={() => {}}>Select</Menu.Item>
        <Menu.Item icon={icons ? <FolderPlus /> : undefined} onSelect={() => {}}>New folder</Menu.Item>
        <Menu.Item icon={icons ? <FileDocument /> : undefined} onSelect={() => {}}>Scan documents</Menu.Item>
        <Menu.Item icon={icons ? <Desktop /> : undefined} onSelect={() => {}}>Connect to server</Menu.Item>
        <Menu.Separator />
        {checkmarks && (
          <>
            <Menu.Item icon={icons ? <VideoGrid /> : undefined} checked={view === "icons"} onSelect={() => setView("icons")}>Icons</Menu.Item>
            <Menu.Item icon={icons ? <List /> : undefined} checked={view === "list"} onSelect={() => setView("list")}>List</Menu.Item>
            <Menu.Separator />
            {SORTS.map(([value, label]) => (
              <Menu.Item
                key={value}
                checked={sort === value}
                description={value === "date" && sort === "date" ? "Newest to oldest" : undefined}
                onSelect={() => setSort(value)}
              >
                {label}
              </Menu.Item>
            ))}
            <Menu.Separator />
            <Menu.Sub>
              <Menu.SubTrigger>Group by</Menu.SubTrigger>
              <Menu.SubContent>
                {GROUPS.map(([value, label]) => (
                  <Menu.Item key={value} checked={group === value} onSelect={() => setGroup(value)}>{label}</Menu.Item>
                ))}
              </Menu.SubContent>
            </Menu.Sub>
          </>
        )}
        <Menu.Item onSelect={() => {}}>View options</Menu.Item>
      </Menu.Content>
    </Menu>
  </Toolbar.Trailing>
</Toolbar>`

export const radioCode = `import { Button, Menu, Toolbar } from "@plexui/react-native"
import { DotsHorizontal } from "@plexui/react-native/icons"

const PERIODS = [
  { value: "any", label: "Any time" },
  { value: "today", label: "Today", disabled: true },
  { value: "last7d", label: "Last 7 days" },
  { value: "last30d", label: "Last 30 days" },
  { value: "last3m", label: "Last 3 months" },
]

const [value, setValue] = useState("any")

<Toolbar titleStyle="large">
  <Toolbar.Title>Radio items</Toolbar.Title>
  <Toolbar.Trailing>
    <Menu>
      <Menu.Trigger>
        <Button color="secondary" variant="soft" uniform accessibilityLabel="More"><DotsHorizontal /></Button>
      </Menu.Trigger>
      <Menu.Content align="end" aria-label="Date modified">
        {PERIODS.map((period) => (
          <Menu.Item
            key={period.value}
            checked={value === period.value}
            disabled={period.disabled}
            onSelect={() => setValue(period.value)}
          >
            {period.label}
          </Menu.Item>
        ))}
      </Menu.Content>
    </Menu>
  </Toolbar.Trailing>
</Toolbar>`

## Examples

### Overview

<ComponentPreview details={<>
  <p>The parts follow <code>@plexui/ui</code>'s Menu: <code>Menu.Trigger</code>, <code>Menu.Content</code>, <code>Menu.Item</code>, <code>Menu.Separator</code>, <code>Menu.Label</code> and <code>Menu.Sub</code> with its <code>Menu.SubTrigger</code> and <code>Menu.SubContent</code>; a radio item is a <code>Menu.Item</code> with <code>checked</code>. On iOS the symbol stands at the leading edge of a row at least 42 tall. On Android the menu is a popup: a white panel (a dark gray at night) with 12dp corners and a soft shadow, as wide as its items (196dp to 288dp), that opens over its trigger, its top on the trigger's top (its bottom on the trigger's bottom when the screen ends first). It unrolls from the trigger's side while its rows fade in one after another, and fades out as it lifts 5dp. A press ripples from the touch point in the list selector (black at 6%, white at 5% at night), clipped to the item and the corners. Long menus scroll, and a tap outside closes the menu.</p>
  <p>In a browser (react-native-web) the iOS menu is nearly opaque, white at 90% (rgb(28, 28, 30) at 90% at night) over a 30px blur, as a CSS blur shows far more of what is under it than iOS draws; the glass of buttons and chips there is gray at 80% (rgb(28, 28, 30) at 70%) over a 20px blur, so a label reads over a photo. A theme sets them with <code>--menu-mobile-web-background</code>, <code>--menu-mobile-web-backdrop-filter</code>, <code>--mobile-glass-web-background</code>, <code>--mobile-glass-web-background-active</code> and <code>--mobile-glass-web-backdrop-filter</code>; a brand that sets <code>--menu-mobile-background</code> or <code>--mobile-glass-background</code> keeps its own.   <p>A folder's menu, as Files opens it on iOS: actions with their icons, then the view and the sort order as options, Date with its order under it (<code>description</code>) while it is the chosen sort, and Group by, a submenu of its own options. Turn off <code>icons</code> for the same menu in plain text, and <code>checkmarks</code> for the folder's actions alone, a shorter menu with no checkmark column.</p>
  <p>An item takes its symbol as <code>icon</code>, in the label's color: 20pt on iOS, centred in a 36pt slot (in a menu with checkmarks it stands 2pt past the labels' line, its label 14 after it), 24dp on Android. An option is an item with <code>checked</code>: a menu whose items set it, true or false, keeps a column for the checkmark on iOS, so every label lines up, and on Android a chosen item shows a thin check in the text color at the end of its row, 18dp after the text. Choosing an item runs its <code>onSelect</code> and closes the menu, an option too.</p>

On iOS the menu is the 250pt panel of iOS menus unless <code>width</code> sets another.</p>

On iOS, as iOS 26 menus, the submenu opens over the menu with its heading on that row, in Semibold with a chevron down, while the menu steps back (a touch smaller, its rows at half strength); a press on the heading or on the menu behind goes back. On Android the submenu slides in from the end, headed by a Back row and an 8dp band, while the menu moves back under a dim and the popup takes the submenu's size; Back, a swipe towards the end or a flick goes back. Choosing an item of a submenu closes the whole menu.</p>

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Menu | Menu |
| Look | A 250pt glass panel with 34pt corners and 17pt rows | A popup: a white (night: dark gray) panel with 12dp corners, 48dp items in 16sp, groups parted by an 8dp band |
| Motion | Grows from its anchor | Unrolls from its trigger as its rows fade in; a submenu slides in, and a swipe takes it back |
| From a list row | Opens at its button | Dims the screen around the row, the popup 8dp under the row at its end |
