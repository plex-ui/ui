export const overviewCode = `import { BottomSheet, Button, Input, Text } from "@plexui/react-native"
import { Email } from "@plexui/react-native/icons"

const [open, setOpen] = useState(false)
const [hero, setHero] = useState(false)
const [title, setTitle] = useState(true)
const [dismissible, setDismissible] = useState(true)
const [email, setEmail] = useState("")

<Button color="secondary" variant="outline" onPress={() => setOpen(true)}>
  Sign in
</Button>
<BottomSheet
  open={open}
  onOpenChange={setOpen}
  title={title ? "Sign in" : undefined}
  // A symbol over a large centered title; the bar keeps Close alone
  hero={hero ? { icon: <Email />, title: "Sign in", description: "We’ll send a code to this address." } : undefined}
  aria-label="Sign in"
  dismissible={dismissible}
>
  <View style={{ gap: 20 }}>
    {!hero && <Text color="secondary">We’ll send a code to this address.</Text>}
    <Input
      accessibilityLabel="Email"
      placeholder="Email address"
      variant="soft"
      size="3xl"
      pill // iOS
      value={email}
      onChangeText={setEmail}
      keyboardType="email-address"
      autoComplete="email"
      autoCapitalize="none"
    />
  </View>
  <BottomSheet.Footer>
    <Button size="xl" block onPress={() => setOpen(false)}>
      Continue
    </Button>
  </BottomSheet.Footer>
</BottomSheet>`

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Sheet | Modal bottom sheet |
| Look | Bar with Close and a centered title | A sheet with 35dp top corners, no handle; Close in a glass circle concentric with the corner and a centered title |

On Android the content and the actions stand 12dp from the sheet's sides, as a page's groups; a screen that keeps its content at Material's compact 16dp sets `--bottom-sheet-android-padding-inline`.
