export const overviewCode = `import { ActionSheet, Button } from "@plexui/react-native"
import { Copy, EditPencil, RemoveTrash } from "@plexui/react-native/icons"

const [open, setOpen] = useState(false)
const [title, setTitle] = useState(true)
const [message, setMessage] = useState(true)
const [destructive, setDestructive] = useState(true)
const [disabled, setDisabled] = useState(false)
const [icons, setIcons] = useState(true) // Android: iOS leaves icons out

<Button color="secondary" variant="outline" onPress={() => setOpen(true)}>
  Project actions
</Button>
<ActionSheet
  open={open}
  onOpenChange={setOpen}
  title={title ? "Website redesign" : undefined}
  message={message ? "What would you like to do with this project?" : undefined}
  actions={[
    { label: "Rename", icon: icons ? <EditPencil /> : undefined, onPress: rename },
    { label: "Duplicate", icon: icons ? <Copy /> : undefined, disabled, onPress: duplicate },
    { label: "Delete project", icon: icons ? <RemoveTrash /> : undefined, destructive, onPress: remove },
  ]}
/>`

export const actionsOnlyCode = `import { CameraPhoto, Folder, Images } from "@plexui/react-native/icons"

const [open, setOpen] = useState(false)

<Button color="secondary" variant="outline" onPress={() => setOpen(true)}>
  Add a photo
</Button>
<ActionSheet
  open={open}
  onOpenChange={setOpen}
  cancelLabel="Cancel"
  actions={[
    { label: "Take photo", icon: <CameraPhoto />, disabled: true },
    { label: "Photo library", icon: <Images />, onPress: openLibrary },
    { label: "Browse files", icon: <Folder />, onPress: browse },
  ]}

### Overview

On Android it is a menu sheet, a <a href="/react-native/components/bottom-sheet">Bottom Sheet</a> with a list of 48dp rows in 16sp, each icon in a 56dp slot at the start (with no icons the labels start 16dp in). ## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Action sheet (confirmation dialog) | Modal bottom sheet with a list |
| Look | Glass capsules, cancel apart | A menu sheet: 48dp rows, each icon in a 56dp slot, no handle |
