export const overviewCode = `import { ProgressSteps } from "@plexui/react-native"
import { Bolt, Email, Globe, Users } from "@plexui/react-native/icons"

const [variant, setVariant] = useState("default")
const [current, setCurrent] = useState(2)
const [size, setSize] = useState("md") // the minimal bar: "md" or "lg"
const [color, setColor] = useState("default")
const [connectorStyle, setConnectorStyle] = useState("solid") // dashes the connectors ahead of current
const [icons, setIcons] = useState(false)
const [text, setText] = useState("full") // "full", "title" or "none"

const steps = [
  { title: "Your details", description: "Please provide your name and email", icon: <Email /> },
  { title: "Company details", description: "A few details about your company", icon: <Globe /> },
  { title: "Invite your team", description: "Start collaborating with your team", icon: <Users /> },
  { title: "Add your socials", description: "Share posts to your social accounts", icon: <Bolt /> },
]

// The minimal bar counts its steps with total
variant === "minimal" ? (
  <ProgressSteps variant="minimal" current={current} total={steps.length} size={size} color={color} />
) : (
  // A phone shows the steps vertically: four columns cannot hold a description
  <ProgressSteps current={current} orientation="vertical" size={size} color={color} connectorStyle={connectorStyle}>
    {steps.map((step) => (
      <ProgressSteps.Step key={step.title} icon={icons ? step.icon : undefined}>
        {/* Without text the title stays for screen readers */}
        <ProgressSteps.Title visuallyHidden={text === "none"}>{step.title}</ProgressSteps.Title>
        {text === "full" && <ProgressSteps.Description>{step.description}</ProgressSteps.Description>}
      </ProgressSteps.Step>
    ))}
  </ProgressSteps>
)`

export const withNavigationCode = `import { Button, ProgressSteps, Text, useTheme } from "@plexui/react-native"
import { View } from "react-native"

const steps = ["Your details", "Company details", "Invite your team", "Add your socials"]
const [step, setStep] = useState(1)
const theme = useTheme()

<View style={{ alignItems: "center", gap: 24 }}>
  <ProgressSteps variant="minimal" current={step} total={4} />
  <View style={{ alignItems: "center", gap: 4 }}>
    <Text accessibilityRole="header" size="lg" weight="semibold">Step {step} of 4</Text>
    <Text size="md" color={theme.color("--color-text-secondary")}>{steps[step - 1]}</Text>
  </View>
  <View style={{ flexDirection: "row", gap: 12 }}>
    <View style={{ width: 112 }}>
      <Button block color="secondary" variant="outline" disabled={step === 1} onPress={() => setStep((s) => Math.max(1, s - 1))}>
        Previous
      </Button>
    </View>
    <View style={{ width: 112 }}>
      <Button block color="primary" disabled={step === 4} onPress={() => setStep((s) => Math.min(4, s + 1))}>
        Next
      </Button>
    </View>
  </View>
</View>`

### Overview

<ComponentPreview details={<>
  <p><code>current</code> is the active step, counted from 1: the steps before it are completed and show a check. <code>variant="minimal"</code> draws a compact bar of segments instead, one per step, and takes the count as <code>total</code>; it has no orientation, connector, icons or text, so those settings are for the default variant, and it draws <code>sm</code> as <code>md</code>, so it offers <code>md</code> and <code>lg</code>. <code>size</code> sets the scale of the indicators, and <code>color</code> the color scheme of completed and active steps. <code>connectorStyle</code> draws the lines ahead of the current step solid or dashed; a completed line is solid, so at the last step the setting is not offered. <code>icon</code> on a <code>ProgressSteps.Step</code> shows a custom icon instead of its number. A step shows a <code>Title</code> and a <code>Description</code>; leave out the description, or show the indicators alone when the step's details are shown elsewhere. Keep the title then for screen readers with <code>visuallyHidden</code>: an icon or a check has no name, and the step takes no room under its indicator.</p>
  <p>On a phone the four columns of a horizontal row cannot hold a step's description, so the steps stand vertically and the example has no <code>orientation</code> setting; <code>orientation="horizontal"</code> (the default) lays them in a row on a tablet. Neither platform has a control for this: iOS draws the Plex steps in the tokens, the ink for completed and active steps; Android draws them in the accent on Material's container gray. A change of step fades the colors in 120ms, at once with reduced motion. On a phone a screen reader reads each step as one element, its title and description, the current one selected. With Navigation shows a compact composition for a phone screen.</p>
</>} code={overviewCode} previewMinHeight={168}>
