export const controllerCode = `import { useEffect, useRef, useState } from "react"
import { Button, Popover, Text, usePopoverController, useTheme } from "@plexui/react-native"
import { Sparkles } from "@plexui/react-native/icons"
import { TextInput, View } from "react-native"

function PromptForm({ value, onChange, submitting, onSubmit }) {
  const theme = useTheme()
  const { shake, close } = usePopoverController()
  return (
    <View style={{ paddingTop: 8 }}>
      <TextInput
        value={value}
        onChangeText={onChange}
        accessibilityLabel="Prompt"
        multiline
        numberOfLines={3}
        placeholder="Describe what you're using the model for, and we'll generate system instructions."
        placeholderTextColor={theme.colors.textTertiary}
        style={{ minHeight: 88, paddingHorizontal: 16, paddingVertical: 8, fontSize: 16, lineHeight: 24, color: theme.colors.text }}
      />
      <View style={{ paddingHorizontal: 12, paddingBottom: 12, flexDirection: "row", alignItems: "center", justifyContent: "space-between" }}>
        <View style={{ flexDirection: "row", alignItems: "center", gap: 6 }}>
          <Sparkles size={16} color={theme.colors.textTertiary} />
          <Text size="sm" color="tertiary">Free beta</Text>
        </View>
        <Button color="primary" size="xs" pill={false} loading={submitting} onPress={() => onSubmit(close, shake)}>
          Create
        </Button>
      </View>
    </View>
  )
}

export function PopoverControllerDemo() {
  const [open, setOpen] = useState(false)
  const [value, setValue] = useState("")
  const [submitting, setSubmitting] = useState(false)
  const timer = useRef(undefined)
  const session = useRef(0)

  useEffect(() => () => {
    session.current++
    clearTimeout(timer.current)
  }, [])

  const changeOpen = (next) => {
    session.current++
    clearTimeout(timer.current)
    timer.current = undefined
    if (next) {
      setSubmitting(false)
      setValue("")
    }
    setOpen(next)
  }

  const submitPrompt = (close, shake) => {
    if (!value) {
      shake()
      return
    }
    if (timer.current !== undefined) return
    setSubmitting(true)
    const current = session.current
    timer.current = setTimeout(() => {
      if (session.current !== current) return
      timer.current = undefined
      close()
    }, 2000)
  }

  return (
    <Popover open={open} onOpenChange={changeOpen}>
      <Popover.Trigger>
        <Button color="primary" pill={false}>Generate</Button>
      </Popover.Trigger>
      <Popover.Content>
        <PromptForm value={value} onChange={setValue} submitting={submitting} onSubmit={submitPrompt} />
      </Popover.Content>
    </Popover>
  )
}`

### Overview

<ComponentPreview details={<>
  <p>Put a <code>Popover.Trigger</code> and a <code>Popover.Content</code> inside a <code>Popover</code>. A tap on the trigger opens the panel and a tap anywhere else closes it; <code>open</code> and <code>onOpenChange</code> control it. The trigger is any element that hands its native view to <code>ref</code> (a Button, a Pressable, a View) and keeps its own place in the layout. <code>side</code> places the panel (<code>bottom</code> by default; it turns when it does not fit) and <code>align</code> lines it up with the start, center or end of the trigger. A phone has no room beside the trigger, so the example opens below it, centered, as the React page's Phone preview does, and a phone has no hover, so there is no <code>showOnHover</code>.</p>
  <p>Size the panel with any combination of <code>width</code>, <code>minWidth</code> and <code>maxWidth</code>: a <code>minWidth</code> of 300 and a <code>maxWidth</code> of 700 by default, capped by the room inside the window and its <code>collisionPadding</code> (20); <code>{'minWidth="auto"'}</code> lets the panel hug its content.</p>
  <p>On iOS the panel is the compact glass panel an iPhone presents a popover as: the glass of the menus with their shadow and edge (<code>--popover-mobile-background</code>, <code>--menu-shadow</code>) and 34pt corners, growing from 0.95 as it fades in. <code>translucent</code> gives it the web's frosted gradient over a 16px blur with its own shadow instead. On Android it is a popup of the Plex Android look: its surface with 12dp corners and a level 2 shadow, popping in from 0.95 with a slight overshoot; Material has no translucent surface, so <code>translucent</code> does nothing there.</p>

### Nesting

A popover opened from inside another stands over it, and Android's back and the web's Escape close the topmost one first.

## API

**Popover** (root)

<PropsTable data={[
  { prop: "open", type: "boolean", description: "Controlled open state; uncontrolled when omitted" },

**Popover.Trigger**: Renders the element that toggles the popover; the panel stands at it and follows it when it moves. Its only child is an element that hands its native view to `ref`.

**Popover.Content**

**usePopoverController()**: Hook available inside `Popover.Content`. Returns `{ close: () => void, shake: () => void }`.

The web's `showOnHover` and `hoverOpenDelay` are for a pointer's hover, `Popover.Anchor` takes a DOM element and `zIndex` orders the page's layers: React Native has none of them (an overlay stands over the app). `style` stands for `className`.
