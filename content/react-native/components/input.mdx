export const overviewCode = `import { Input, List, ListItem, Text } from "@plexui/react-native"

const [value, setValue] = useState("janeappleseed")
const [adornment, setAdornment] = useState<"none" | "start" | "end">("none")
const [clearable, setClearable] = useState(false)
const [selectTextOnFocus, setSelectTextOnFocus] = useState(false)
const [readOnly, setReadOnly] = useState(false)
const [disabled, setDisabled] = useState(false)
const [invalid, setInvalid] = useState(false)

// The input is the value of a row: a ghost field, the row is its frame. iOS clears only a value that starts its
// column (columns) or has a line of its own, never one at the row's end
<List columns={clearable}>
  <ListItem
    label="Username"
    error={invalid ? "This username is taken." : undefined}
    trailing={
      <Input
        variant="ghost"
        value={value}
        onChangeText={setValue}
        onClear={clearable ? () => setValue("") : undefined}
        maxLength={16}
        placeholder="Required"
        autoCapitalize="none"
        startAdornment={adornment === "start" ? <Text color="secondary">@</Text> : undefined}
        endAdornment={adornment === "end" ? <Text color="secondary" tabularNums>{value.length}/16</Text> : undefined}
        selectTextOnFocus={selectTextOnFocus}
        readOnly={readOnly}
        disabled={disabled}
        invalid={invalid}
        accessibilityLabel="Username"
      />
    }
  />
</List>`

export const passwordCode = `import { Input, List, ListItem } from "@plexui/react-native"
import { Eye, EyeOff } from "@plexui/react-native/icons"

const [visible, setVisible] = useState(false)

<List>
  <ListItem
    label="Password"
    trailing={
      <Input
        variant="ghost"
        placeholder="Required"
        secureTextEntry={!visible}
        autoCapitalize="none"
        autoComplete="password"
        accessibilityLabel="Password"
        endAdornment={
          <Input.AdornmentButton
            accessibilityLabel={visible ? "Hide" : "Show"}
            selected={visible}
            onPress={() => setVisible((value) => !value)}
          >
            {visible ? <EyeOff /> : <Eye />}
          </Input.AdornmentButton>
        }
      />
    }
  />
</List>`

export const ssnCode = `import { Input, List, ListItem } from "@plexui/react-native"
import { Eye, EyeOff } from "@plexui/react-native/icons"

const [digits, setDigits] = useState("")

const formatted = formatSSN(digits) // XXX-XX-XXXX

<List>

    label="SSN"

        accessibilityLabel="SSN"

            selected={visible}
            onPress={() => setVisible((value) => !value)}
          >
            {visible ? <EyeOff /> : <Eye />}
          </Input.AdornmentButton>
        }
      />
    }
  />
</List>`

<ComponentPreview details={<>
  <p>On a phone, put an input in a row of a <code>List</code>, the grouped section of a settings form: the row sets the label and its height, and the input is a <code>ghost</code> field at its end, which draws no frame of its own. The props follow <code>@plexui/ui</code>'s Input, on top of React Native's <code>TextInput</code>, whose other props pass through.</p>
  <p><code>variant</code> styles the field. On iOS <code>outline</code> draws a border that darkens on focus, <code>soft</code> a gray fill with no focus ring, and <code>ghost</code> neither, for a field its container already frames, such as a row; it still turns red when <code>invalid</code>. On Android <code>outline</code> is the outlined field, 8dp corners and a hairline that thickens to 1.67dp in the accent on focus (its lighter text at night); <code>soft</code> a field filled with the text color at 5%, with no line; and <code>ghost</code> the bare field of a card, with neither, its text 21dp in. Their text is 17sp with a hint at 38%; the caret takes the accent, and the text color in a <code>ghost</code> field. A <code>soft</code> field with a <code>startAdornment</code> and no label is the search field: a 40dp pill of the text color at 5% (7% at night), 15sp text 48dp in after the 24dp magnifier, its hint at 50%. <code>size</code> is the Plex UI scale: on iOS every step is the 44pt phone control and the size changes the radius, gutter and type; Android fields are 50dp, 56 with a label. <code>pill</code> makes a capsule; on iOS one 48pt or taller sets its text 20pt in, as Mail's and Apple Account's fields. In a <code>Toolbar</code> the field takes the bar's look: glass on iOS, and on Android the search of a bar, a capsule of the bar's glass at its 46dp height, 18sp text 18dp in, the hint in a field's gray and the caret in the text color, in the bar's row or in a <code>Toolbar.Row</code> under it.</p>
  <p><code>startAdornment</code> puts content before the value, such as a prefix or a search icon, and <code>endAdornment</code> after it, such as a character counter; an icon takes the platform's size and color. <code>onClear</code> shows a clear button while the field has a value and is neither disabled nor read-only: on iOS the filled circle of a text field, while it is edited (UITextField's whileEditing), or while a search field (<code>inputMode="search"</code>, a bar's field) has text; a value at a row's end takes none, as in Settings and Contacts: only a value that starts its column (a List with <code>columns</code>, as here once <code>clearable</code> is on) or stands on its own line under the label does; there its 44pt target reaches over and under the row, which keeps its height, and the circle ends at the row's end inset. On Android Material's clear-text icon (a filled circle with the cross cut out, in <code>onSurfaceVariant</code>), while the field has focus and a value, as Material Components' clear-text end icon; a press keeps the field editing; in a search field on Android, a cross that stays in place and shows with the first letter, growing and turning in (from 45° over 180ms in a bar, from 90° over 380ms in the 40dp field). A press clears the text (<code>onChangeText</code> receives an empty string), then calls it.</p>
  <p><code>selectTextOnFocus</code>, React Native's own prop, selects all the text when the field takes focus, as the web's <code>autoSelect</code> does for copy actions. <code>readOnly</code> keeps the value and hides the clear button. <code>disabled</code> dims the field and takes no input. <code>invalid</code> draws it in the error color: a red border on iOS; on Android a red line (a red outline on a soft field, a 2dp red underline on a ghost one) and a red label, caret and trailing icon. The error message here is the row's <code>description</code>, under its label.</p>
  <p>As the row's <code>ghost</code> field, in its <code>trailing</code> slot, the example offers neither <code>variant</code> nor <code>size</code>. The React page's <code>orientation</code> and <code>columns</code> belong to its form rows, which React Native's List does not have, and browser autofill extensions are a desktop matter: iOS and Android offer autofill from the system, by <code>autoComplete</code>.</p>

`TextField` is an Input with a `label`, a `description` under it and an `errorMessage` that takes the description's place and marks the field invalid; `required` adds an asterisk to the label. On iOS the label stands over the field (15/20 medium) and the help 4pt under it; on Android the label floats onto the outline, or to the top of a filled field, when the field has focus or a value, in the accent while it has focus (its lighter text at night), and the help is 14sp text in the secondary gray under the field.

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Text field | Text field (filled or outlined) |
| Look | A 44pt field in 17pt type, its border darkening on focus | A 50dp field in 17sp type, its outline thickening in the accent on focus |
| In a List row | The row's value: no frame, the text at the row's end, the error a line under the row | A field at the row's end, the error under the label |
