export const overviewCode = `import { Button, toast } from "@plexui/react-native"

// Each type's title, its short description, a long one that wraps to two lines, and its action's label
const messages = {
  default: { title: "Changes saved", description: "Your profile is live for everyone on the team.", long: "Your profile is live for everyone on the team. Anna Kim and 3 others will see your new photo and title.", action: "Undo" },
  success: { title: "Inquiry ID copied", description: "Paste it into the support form.", long: "Paste it into the support form so we can find your case. We usually reply within one business day.", action: "Open" },
  error: { title: "Failed to assign owner", description: "The owner has left the workspace.", long: "Anna Kim has left the workspace. Choose another owner for “Q3 roadmap” or invite her back first.", action: "Retry" },
  warning: { title: "Session expires in 2 minutes", description: "Unsaved changes will be lost.", long: "Unsaved changes to “Q3 roadmap” will be lost. Stay signed in to keep editing.", action: "Stay" },
  info: { title: "New version available", description: "Reload to get the latest features.", long: "Version 2.4 adds shared folders and faster search. Reload to get the latest features.", action: "Reload" },
}

const [type, setType] = useState("default") // or "success", "error", "warning", "info"
const [icon, setIcon] = useState(true) // the type's icon; false passes icon: false
const [description, setDescription] = useState("none") // or "short", "long"
const [action, setAction] = useState("none") // or "action"
const [closeButton, setCloseButton] = useState(false) // Android
const [persistent, setPersistent] = useState(false) // with an action or a close button

const message = messages[type]

const show = () => {
  const options = {
    description: description === "none" ? undefined : description === "short" ? message.description : message.long,
    action: action === "none" ? undefined : { label: message.action, onClick: () => {} },
    closeButton,
    // Open until closed, once there is a button to close it
    duration: persistent && (action !== "none" || closeButton) ? Infinity : undefined,
    icon: icon ? undefined : false,
  }
  ;(type === "default" ? toast : toast[type])(message.title, options)
}

<Button color="secondary" variant="outline" onPress={show}>
  Show toast
</Button>`

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | None (a Plex banner; iOS has no system toast) | None (a Plex view drawn as a dark card) |
| Look | Glass banner from the top | A dark card with 16dp corners at the bottom |

Neither platform's system toast is used: on Android the toast is a view of the app, not Material's Snackbar.
