import { RNExample } from "@/components/docs/react-native/RNExample"

export const overviewCode = `import { List, ListItem, Stepper } from "@plexui/react-native"

const [disabled, setDisabled] = useState(false)
const [showValue, setShowValue] = useState(false)
const [copies, setCopies] = useState(1)
const [nights, setNights] = useState(3)

<List footer="Up to 20 copies and 14 nights.">
  <ListItem
    label={showValue ? "Copies" : copies === 1 ? "1 copy" : \`\${copies} copies\`}
    trailing={<Stepper aria-label="Copies" value={copies} onChange={setCopies} min={1} max={20} disabled={disabled} showValue={showValue} />}
  />
  <ListItem
    label={showValue ? "Nights" : nights === 1 ? "1 night" : \`\${nights} nights\`}
    trailing={<Stepper aria-label="Nights" value={nights} onChange={setNights} min={1} max={14} disabled={disabled} showValue={showValue} />}
  />
</List>`

export const rangeCode = `import { List, ListItem, Stepper } from "@plexui/react-native"
import { View } from "react-native"

const [wrap, setWrap] = useState(true)
const [opacity, setOpacity] = useState(0.5)
const [hour, setHour] = useState(23)

<View style={{ gap: 35 }}>
  <List footer="Steps of 10% from 0% to 100%.">
    <ListItem
      label={\`Opacity: \${Math.round(opacity * 100)}%\`}
      trailing={<Stepper aria-label="Opacity" value={opacity} onChange={setOpacity} min={0} max={1} step={0.1} />}
    />
  </List>
  <List footer={wrap ? "Past 23:00 the hour goes on from 00:00, and neither segment stops." : "The hour stops at 00:00 and 23:00, where a segment turns tertiary."}>
    <ListItem
      label={\`Hour: \${String(hour).padStart(2, "0")}:00\`}
      trailing={<Stepper aria-label="Hour" value={hour} onChange={setHour} min={0} max={23} wrap={wrap} />}
    />
  </List>
</View>`

export const cartCode = `import { List, ListItem, Stepper } from "@plexui/react-native"

<List footer="At one, the minus is a trash can. Orange Juice has 3 left and Cheesecake 1.">
  {items.map((item) => (
    <ListItem
      key={item.id}
      label={item.name}
      trailing={
        <Stepper
          aria-label={item.name}
          value={item.quantity}
          onChange={(quantity) => setQuantity(item.id, quantity)}
          min={1}
          max={item.max}
          showValue
          onRemove={() => remove(item.id)}
          removeLabel={\`Remove \${item.name}\`}
        />
      }
    />
  ))}
</List>`

### Overview

<ComponentPreview details={<>
  <p>A stepper changes a value by small amounts and shows no value of its own. For values that vary widely, pair it with an input. On iOS it is the 93 × 32 capsule of a native iOS 27.2 stepper: two 46.5pt segments, each with its own fill, a short separator between them, 22pt symbols and a 44pt target; a held segment darkens at once and fades back in 150ms.</p>
  <p>Put a stepper at the end of a <code>ListItem</code> and name its value in the row's label, such as 3 copies: the stepper shows no value of its own, so give it an <code>aria-label</code>. At the lowest value the minus turns tertiary and stops stepping, at the highest the plus, as 1 copy shows; the segment keeps its fill, as in iOS. Holding a segment keeps stepping, after 400ms every 100ms, as a native stepper does. A disabled stepper shows both symbols tertiary. With <code>showValue</code> the value stands between the minus and the plus on one fill, as a cart's quantity does, and the row's label only names it, such as Copies: a 96 × 36 capsule with round 36pt segments and the value at 15pt medium. The stepper is always the capsule on a phone, so the example does not offer <code>pill</code>.</p>
  <p>Android has no stepper of its own: it draws the same capsule in Material's colors, Material's container gray with the pressed state layer, the symbols in the text color (at 38% when they cannot act) and a 48dp target. For VoiceOver and TalkBack the stepper is one adjustable element with its value: swipe up or down to step.</p>
</>} code={overviewCode} previewMinHeight={240}>
