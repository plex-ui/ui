export const overviewCode = `import { Button, Tooltip } from "@plexui/react-native"
import { Copy } from "@plexui/react-native/icons"

const [trigger, setTrigger] = useState("text") // or "button", "icon"
const [compact, setCompact] = useState(false)
const [side, setSide] = useState("top") // or "right", "bottom", "left"
const [disabled, setDisabled] = useState(false)

<>
  {trigger === "text" && (
    // Tooltip.TriggerDecorator makes text a trigger
    <Tooltip content="This is additional context that appears when the trigger is hovered or focused" compact={compact} side={side}>
      <Tooltip.TriggerDecorator size="md">Simple text with tooltip</Tooltip.TriggerDecorator>
    </Tooltip>
  )}
  {trigger === "button" && (
    <Tooltip content={disabled ? "This action is disabled for reasons" : "Tooltip on button"} compact={compact} side={side}>
      <Button color="primary" size="lg" disabled={disabled}>
        Sample button
      </Button>
    </Tooltip>
  )}
  {trigger === "icon" && (
    <Tooltip content={disabled ? "This action is disabled for reasons" : "Copy to clipboard"} compact={compact} side={side}>
      <Button color="secondary" variant="ghost" size="md" accessibilityLabel="Copy to clipboard" disabled={disabled}>
        <Copy />
      </Button>
    </Tooltip>
  )}
</>`

### Overview

<ComponentPreview details={<>
  <p>Tooltips give supplementary information. A phone has no hover: a long press on the trigger shows the tooltip for a second and a half, as Android shows a tooltip, and a tap anywhere hides it sooner; <code>forceOpen</code> keeps it shown or hidden. Keep what people need in view as visible helper text, or open a <a href="/react-native/components/popover">Popover</a> on a tap. A screen reader reads the content as the trigger's description, so it is never lost.</p>
  <p>Any element that hands its native view to <code>ref</code> can be the trigger, and it keeps its place in the layout: <code>trigger</code> shows text, a button and an icon button. Wrap text in <code>Tooltip.TriggerDecorator</code>, which a tap opens too and which shows the dotted underline of an open tooltip. An icon button names itself with <code>accessibilityLabel</code>, and its tooltip says the same in words. A disabled button tells why in its tooltip: its long press still shows it. <code>content={'{null}'}</code> shows none.</p>
  <p>On iOS, which has no tooltips, it is the web's tooltip: the elevated surface with its shadow, 8pt corners and 14/20 text, 12 by 16 in (<code>gutterSize</code>); <code>compact</code>, for one to three words, is inverted, 2 by 8 in. On Android it is Material 3's: <code>compact</code> the plain tooltip (the inverse surface, Body Small, 4dp corners), otherwise the rich tooltip's container (12dp corners, Body Medium, a level 2 shadow), 4dp from the trigger. <code>side</code> is the side it prefers; it turns to the other side when there is no room.</p>

### Tooltip

`Tooltip.TriggerDecorator` turns text into a trigger that a tap or a long press opens (`focusable`, default `true`, makes it a stop of the web's keyboard). It is a `Text` of its own, so it takes `variant`, `size`, `weight`, `color` and `style` for the type the web's decorator inherits from the text around it. The web's `openDelay`, `interactive` and `preventUnintentionalClickToClose` are for a pointer's hover and have no counterpart; `contentStyle` stands for `contentClassName`.
