import { RNExample } from "@/components/docs/react-native/RNExample"

export const overviewCode = `import { Button, List, Slider, Text } from "@plexui/react-native"

const scaleMarks = [0, 25, 50, 75, 100].map((value) => ({ value, label: String(value) }))
// The slider is named after what it sets: a value in percent, in dollars or bare; a range or tiers of them
const names = {
  one: { none: "Brightness", unit: "Volume", prefixUnit: "Max price" },
  two: { none: "Age range", unit: "Humidity range", prefixUnit: "Price range" },
  three: { none: "Age groups", unit: "Discount tiers", prefixUnit: "Price tiers" },
}

const [thumbs, setThumbs] = useState("one") // or "two", "three"
const [values, setValues] = useState({ one: 50, two: [25, 75], three: [10, 40, 70] })
const [minStepsBetweenThumbs, setMinStepsBetweenThumbs] = useState(5) // with two or three thumbs
const [step, setStep] = useState("1") // or "5", "10", "25"; with one thumb
const [resetValue, setResetValue] = useState(true) // with one thumb
const [unit, setUnit] = useState("unit") // or "none", "prefixUnit"
const [marks, setMarks] = useState(false)
const [colors, setColors] = useState(false)
const [adornments, setAdornments] = useState(false)
const [disabled, setDisabled] = useState(false)

// A small and a large letter at the ends size text
const name = adornments && thumbs === "one" ? "Text size" : names[thumbs][unit]
const text = (value) => (unit === "prefixUnit" ? "$" : "") + value + (unit === "unit" ? "%" : "")
const shared = {
  min: 0,
  max: 100,
  step: thumbs === "one" ? Number(step) : 1,
  marks: marks ? scaleMarks : undefined,
  // One hue: the fill in green and the track in a light tint of it
  trackColor: colors ? "#61C45433" : undefined,
  rangeColor: colors ? "#61C454" : undefined,
  // A small and a large letter at the ends, as Settings draws a text size slider
  startAdornment: adornments ? <Text variant="subheadline">A</Text> : undefined,
  endAdornment: adornments ? <Text variant="title2">A</Text> : undefined,
  disabled,
  "aria-label": name,
}

// The list's header names the slider; one value ends the cell, a range goes under the track, and marks show the
// scale in its place. The reset is a row of its own under the list.
<View style={{ gap: 35 }}>
  <List header={name}>
    {thumbs === "one" && !marks ? (
      <View style={{ flexDirection: "row", alignItems: "center", gap: 12, paddingHorizontal: 20, minHeight: 52 }}>
        <View style={{ flex: 1 }}>
          <Slider {...shared} value={values.one} onChange={(one) => setValues({ ...values, one })} />
        </View>
        {/* The letters stand where the value would */}
        {!adornments && (
          <View style={{ minWidth: 43 }}>
            <Text tabularNums align="right" color={disabled ? "disabled" : "secondary"}>{text(values.one)}</Text>
          </View>
        )}
      </View>
    ) : (
      <View style={{ paddingHorizontal: 20, paddingBottom: 4 }}>
        {thumbs === "one" ? (
          <Slider {...shared} value={values.one} onChange={(one) => setValues({ ...values, one })} />
        ) : (
          <Slider {...shared} range value={values[thumbs]} onChange={(value) => setValues({ ...values, [thumbs]: value })} minStepsBetweenThumbs={minStepsBetweenThumbs} />
        )}
        {!marks && (
          <Text variant="subheadline" align="center" tabularNums color={disabled ? "disabled" : "primary"}>
            {values[thumbs].map(text).join(" – ")}
          </Text>
        )}
      </View>
    )}
  </List>
  {thumbs === "one" && resetValue && (
    <List>
      <View style={{ flexDirection: "row", alignItems: "center", paddingHorizontal: 20, minHeight: 52 }}>
        <View style={{ flex: 1 }}>
          <Button block color="info" variant="ghost" disabled={disabled} onPress={() => setValues({ ...values, one: 50 })}>
            Reset to default
          </Button>
        </View>
      </View>
    </List>
  )}
</View>`

export const settingCode = `const [zoom, setZoom] = useState(100)

<List header="Zoom" footer="Makes text and images on the page larger or smaller.">
  <View style={{ flexDirection: "row", alignItems: "center", gap: 12, paddingHorizontal: 20, minHeight: 52 }}>
    <View style={{ flex: 1 }}>
      <Slider aria-label="Zoom" min={50} max={200} step={10} value={zoom} onChange={setZoom} />
    </View>
    <View style={{ minWidth: 43 }}>
      <Text tabularNums align="right" color="secondary">{zoom}%</Text>
    </View>
  </View>
</List>`

### Overview

<ComponentPreview details={<>
  <p>On a phone, each slider stands in a cell of a <code>List</code>, as in iOS Settings: the list's header names the slider and the slider fills the cell. One value ends the cell, a range goes under the track, and with marks the cell shows the scale instead of a value. A reset is a row of its own under the list. Outside a list a slider takes a <code>label</code> of its own, with the value and a round reset at its end.</p>
  <p>Set <code>range</code> and give <code>value</code> two numbers for a range, or three or more for several thumbs; <code>onChange</code> then receives an array. <code>step</code> is how far a drag moves the value at a time: 25 here stops it on the marks. <code>minStepsBetweenThumbs</code> keeps the thumbs that many steps apart, and a thumb never passes its neighbor. A press anywhere on the track moves the nearest thumb there, and each thumb is an adjustable element for VoiceOver and TalkBack.</p>
  <p>On iOS the slider is the phone layer's: a 6pt track filled in the range color, a 38x24 white capsule thumb with a soft shadow, in a 52pt row. On Android it is a 3dp track in a light gray, the part up to the thumb in the accent, and a 12dp round thumb of the accent that grows to 16dp (1dp of radius every 60ms) with a soft halo while held and glides to a new step in 60ms; steps snap on a plain track, and a range fills the track between its thumbs. With <code>ticks</code> the track is a stepped picker: a 6dp dot at each step, the chosen one 12dp, 2dp bars between them, the dots up to the thumb in the accent and the rest gray. A disabled slider fades to half.</p>
  <p>The cell ends with the value, in the secondary label; a range writes its values under the track. <code>marks</code> label points of the scale under the track (on Android over it in 13dp, the one a thumb stands on in the accent, as in the stepped picker) and take the value's place, so with marks the example shows no unit either. <code>trackColor</code> and <code>rangeColor</code> color the track and its fill. <code>startAdornment</code> and <code>endAdornment</code> stand at the ends of the track, as Settings puts a small and a large letter at the ends of a text size slider, and take the place of the value. They are decoration, so the slider's <code>aria-label</code> has to say what it sets.</p>
  <p>The cell has no label row for a reset button, so the reset is a row of its own under the slider's list, its label centered in the accent, as Settings resets a size; a disabled slider dims the row with it. A slider with a <code>label</code> of its own takes <code>resetValue</code> instead, a small round button beside its value. A phone list has no tooltip or vertical slider, so the example leaves those out; <code>orientation="vertical"</code> still stands a slider up outside a list.</p>
</>} code={overviewCode} previewMinHeight={168}>
