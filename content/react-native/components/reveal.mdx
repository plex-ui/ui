export const overviewCode = `import { Card, Reveal } from "@plexui/react-native"

<Reveal
  gap={12}[[#order=reversed]]
  order={[2, 1, 0]}[[/order=reversed]][[#timing=one by one]]
  stagger={500}
  duration={500}[[/timing=one by one]]
  onDone={() => setDone(true)}
>
  {items.map((item) => (
    <Card key={item.title}>
      <Card.Header>
        <Card.Title>{item.title}</Card.Title>
        <Card.Description>{item.text}</Card.Description>
      </Card.Header>
    </Card>
  ))}
</Reveal>`

### Overview

<ComponentPreview details={<>
  <p>Reveal brings its items in one after another, each fading in and rising a little (8 pt), as a list&apos;s rows come in on iOS and Material&apos;s lists enter. By default each starts 50 ms after the one before, on the system&apos;s smooth spring (about 0.35 s).</p>
  <p><code>order</code> sets which comes first, by the items&apos; indexes. <code>stagger</code>, <code>delay</code> and <code>duration</code> set the timing: a <code>stagger</code> as long as the <code>duration</code> plays them one after the other, as <code>timing</code> one by one does here. <code>onDone</code> is told when the last is in, for what follows (a button fading in). <code>play=&#123;false&#125;</code> holds them unseen until the screen is ready.</p>
  <p>A screen reader finds every item from the start. With reduced motion they are all there at once, and <code>onDone</code> is told at once.</p>
</>} code={overviewCode} codeValues={{ timing: "stagger", order: "in order" }} previewMinHeight={420}>

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | None: SwiftUI's opacity and offset transitions on the smooth spring | None: Material's enter motion for a list |
| Look | Items fading in and rising, 50 ms apart | The same |
