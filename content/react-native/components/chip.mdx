export const overviewCode = `import { Button, Chip } from "@plexui/react-native"
import { AtSign, AvatarProfile, Email, Star } from "@plexui/react-native/icons"

const CHIPS = [
  { value: "unread", label: "Unread", icon: <Email /> },
  { value: "starred", label: "Starred", icon: <Star /> },
  { value: "mentions", label: "Mentions", icon: <AtSign /> },
]
// Input chips: the recipients of a message
const RECIPIENTS = [
  { value: "anna", label: "Anna Kim", icon: <AvatarProfile /> },
  { value: "ben", label: "Ben Carter", icon: <AvatarProfile /> },
  { value: "maya", label: "Maya Patel", icon: <AvatarProfile /> },
]

const [type, setType] = useState<"filter" | "action" | "input">("filter") // which of the props below the chips set
const [icon, setIcon] = useState(false)
const [disabled, setDisabled] = useState(false)
const [selected, setSelected] = useState(["unread"])
const [shown, setShown] = useState(RECIPIENTS.map((chip) => chip.value))

<View style={{ flexDirection: "row", flexWrap: "wrap", justifyContent: "center", columnGap: 8 }}>
  {(type === "input" ? RECIPIENTS.filter((chip) => shown.includes(chip.value)) : CHIPS).map((chip) => (
    <Chip
      key={chip.value}
      icon={icon ? chip.icon : undefined}
      disabled={disabled}
      selected={type === "filter" ? selected.includes(chip.value) : undefined}
      onSelectedChange={type === "filter" ? (on) => setSelected((list) => (on ? [...list, chip.value] : list.filter((value) => value !== chip.value))) : undefined}
      onPress={type === "action" ? () => markAs(chip.value) : undefined}
      onRemove={type === "input" ? () => setShown((list) => list.filter((value) => value !== chip.value)) : undefined}
      removeLabel={\`Remove \${chip.label}\`}
    >
      {chip.label}
    </Chip>
  ))}
</View>
{type === "input" && shown.length === 0 && (
  <Button color="secondary" variant="outline" size="sm" onPress={() => setShown(RECIPIENTS.map((chip) => chip.value))}>
    Show again
  </Button>
)}`

### Overview

<ComponentPreview details={<>
  <p>Chip's examples follow the Toggle Group page's, and its props follow the toggles of <code>@plexui/ui</code> (<code>selected</code>, <code>onSelectedChange</code>) and Badge's <code>onRemove</code>. iOS has no chip control, so a chip is drawn as the filter capsules of iOS apps: a 34pt capsule in 15/20 medium on the soft gray fill, filled in the Plex primary when selected, as a selected Button. On Android a chip is a 32dp capsule filled with the text color at 5%, its label 14sp Regular, pressed in the press color of a List row (black at 6%, white at 5% in dark mode); a selected filter chip fills in the Plex primary, as on iOS. On both, the chip stands in a 44pt (iOS) or 48dp (Android) touch target.</p>
  <p>Chips of text take the width of their label, 8 points apart, here the filters of a mailbox that can be on together, as the Toggle Group's Text example. <code>type</code> says which props they take. A chip with <code>selected</code>, <code>defaultSelected</code> or <code>onSelectedChange</code> is a filter chip: a press toggles it, and VoiceOver and TalkBack read it as a checkbox. A chip with <code>onPress</code> alone is an action chip (Material's assist chip): it runs the action and keeps no state. <code>onRemove</code> makes an input chip: a cross after the label, a button of its own named by <code>removeLabel</code>; here the chips are the recipients of a message, the cross removes one, and Show again brings them back.</p>
  <p><code>icon</code> puts a symbol before the label, 18dp on Android. <code>disabled</code> dims a chip (on Android to half, its fill with it; on iOS the label of an off chip turns the disabled gray and a selected chip's primary fades to 40%) and takes no presses; a selected chip stays selected. On iOS a chip is a capsule as wide as its label, selected in the Plex primary; on Android a 32dp capsule, selected in the Plex primary too.</p>
