export const overviewCode = `import { Button, Tip } from "@plexui/react-native"
import { Bell } from "@plexui/react-native/icons"
import { View } from "react-native"

const [shown, setShown] = useState(true)
const [icon, setIcon] = useState(true)
const [onClose, setOnClose] = useState(true)
const [actions, setActions] = useState("none") // or "one", "two"
const [mobileActions, setMobileActions] = useState("rows") // or "buttons"
const [arrowEdge, setArrowEdge] = useState("none") // or "top", "bottom"
// With an arrow, the control the tip explains stands on that side
const control = arrowEdge !== "none" && (
  <Button color="secondary" variant="soft" uniform pill accessibilityLabel="Notifications" style={{ alignSelf: "center" }}>
    <Bell />
  </Button>
)

<View style={{ alignItems: "center", gap: 4 }}>
  {arrowEdge === "top" && control}
  {shown ? (
    <Tip
      icon={icon ? <Bell /> : undefined}
      title="Turn on notifications"
      description="Get a message when someone replies to your post or mentions you."
      onClose={onClose ? () => setShown(false) : undefined}
      actions={actions === "none" ? undefined : (
        <>
          {actions === "two" && <Button color="primary" variant="soft" size="sm" pill>Not now</Button>}
          <Button color="primary" variant={actions === "two" ? "solid" : "soft"} size="sm" pill>Turn on</Button>
        </>
      )}
      mobileActions={actions === "none" ? undefined : mobileActions}
      arrowEdge={arrowEdge === "none" ? undefined : arrowEdge}
    />
  ) : (
    <Button color="secondary" variant="soft" onPress={() => setShown(true)}>Show the tip again</Button>
  )}
  {arrowEdge === "bottom" && control}
</View>`

export const popoverCode = `import { Button, Tip } from "@plexui/react-native"

>
  <Button color="secondary" variant="soft" onPress={() => setOpen(!open)}>
    <Filter />
    Filter
  </Button>
</Tip.Popover>`

### Tip.Popover

Takes `title`, `description`, `icon`, `actions` and `mobileActions` as Tip does, and the control it explains as its only child: an element that hands its native view to `ref` (a Button, a Pressable, a View), which keeps its own place in the layout.

An inline tip is a note named by its title; a popover tip is a dialog named by its title that does not move the focus. A tap outside a popover tip closes it without reaching what is under it.
