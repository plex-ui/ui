import { Flask } from "@plexui/react-native/icons"

const [color, setColor] = useState<"secondary" | "success" | "warning" | "danger" | "info" | "discovery" | "caution">("info")
const [variant, setVariant] = useState<"soft" | "solid" | "outline" | "counter">("soft")
const [size, setSize] = useState<"sm" | "md" | "lg" | "xl">("md")
const [pill, setPill] = useState(false)
const [leading, setLeading] = useState<"none" | "icon">("none")
const [onRemove, setOnRemove] = useState(false)
const [shown, setShown] = useState(true)
// A counter shows a count; a badge with a cross is a tag the reader added
const counter = variant === "counter"
const label = counter ? "12" : (onRemove ? { none: "Design", icon: "Research" } : { none: "New", icon: "Beta" })[leading]

{shown ? (
  <Badge
    color={color}
    variant={variant}
    size={size}
    pill={pill}
    removeLabel={\`Remove \${label}\`}
    onRemove={onRemove && !counter ? () => setShown(false) : undefined}
  >
    {leading === "icon" && !counter && <Flask />}
    {label}
  </Badge>
) : (
  <Button color="secondary" variant="outline" size="sm" onPress={() => setShown(true)}>
    Show again
  </Button>
)}`

## Platforms

| | iOS 26 | Android |
| --- | --- | --- |
| Native component | Badge | Badge |
| Look | The Plex label: a soft, solid or outline fill, or a counter | The same label, in the same tokens |

Both platforms' own badges are a count or a dot on a tab or an icon (TabBar draws those); this Badge is the text label, drawn alike on both.
