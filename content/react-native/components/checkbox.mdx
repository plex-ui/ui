export const overviewCode = `import { Checkbox, ThemeProvider } from "@plexui/react-native"

const [variant, setVariant] = useState<"solid" | "ghost">("solid")
const [label, setLabel] = useState(true)
const [description, setDescription] = useState(false) // with a label
const [orientation, setOrientation] = useState<"left" | "right">("left") // with a label
const [disabled, setDisabled] = useState(false)
// The app's on color, set once on its ThemeProvider: "primary" (the default) or "accent"
const [selectionColor, setSelectionColor] = useState<"primary" | "accent">("primary")

const options = [
  { label: "Photos", description: "Back up photos and videos over Wi-Fi.", defaultChecked: true },
  { label: "Messages", description: "Include chats and attachments in each backup.", defaultChecked: false },
  { label: "Folders", description: "3 of 5 folders are backed up.", defaultChecked: "indeterminate" },
]

<ThemeProvider selectionColor={selectionColor}>
  <View style={label ? { alignSelf: "center", maxWidth: 384, gap: description ? 16 : 12 } : { alignSelf: "center", flexDirection: "row", gap: 16 }}>
    {options.map((option) => (
      <Checkbox
        key={option.label}
        variant={variant}
        disabled={disabled}
        defaultChecked={option.defaultChecked}
        {...(label
          ? { label: option.label, description: description ? option.description : undefined, orientation }
          : { accessibilityLabel: option.label })}
      />
    ))}
  </View>
</ThemeProvider>`

export const selectAllCode = `const [items, setItems] = useState([
  { id: "photos", label: "Photos", checked: true },
  { id: "contacts", label: "Contacts", checked: true },
  { id: "messages", label: "Messages", checked: false },
])

const allChecked = items.every((i) => i.checked)
const noneChecked = items.every((i) => !i.checked)
const parentChecked = allChecked ? true : noneChecked ? false : "indeterminate"

<View style={{ alignSelf: "center", gap: 12 }}>
  <Checkbox
    label="Back up"
    checked={parentChecked}
    onCheckedChange={() => {
      const next = !allChecked
      setItems((prev) => prev.map((i) => ({ ...i, checked: next })))
    }}
  />
  <View style={{ paddingStart: 24, gap: 12 }}>
    {items.map((item) => (
      <Checkbox
        key={item.id}
        label={item.label}
        checked={item.checked}
        onCheckedChange={(next) => setItems((prev) => prev.map((i) => (i.id === item.id ? { ...i, checked: next } : i)))}
      />
    ))}
  </View>
</View>`

### Overview

<ComponentPreview details={<>
  <p>On a phone a single checkbox stays next to its label, while a list of options goes in the rows of a <code>List</code>: each <code>ListItem</code> names an option and ends with the checkbox, as in an iOS settings list. Ending the row with it is the pattern of these docs on both platforms, and on Android the box then stands 21dp in from the card's edge.</p>
  <p>A checkbox is checked, unchecked or indeterminate. Pass <code>checked="indeterminate"</code>, or <code>defaultChecked="indeterminate"</code>, for a partly selected state: a "select all" box whose items are only partly checked, as in <a href="#group-with-select-all">Group with select all</a>. A press on an indeterminate box checks it, so <code>onCheckedChange</code> receives <code>true</code>.</p>
  <p>On iOS a <code>solid</code> box (the default) is the 22pt selection circle of a list, a 1.5pt ring that fills in the selection color with a check; <code>ghost</code> draws the checkmark alone, as an iOS list row marks its chosen rows, in the text color, or in the accent when the selection color is the accent. The selection color is the <code>ThemeProvider</code>&apos;s <code>selectionColor</code>: the Plex primary unless set to the accent, on both platforms. On Android a checkbox is a square box: 18dp with 4dp corners and a gray 1.33dp outline, filled in the selection color when checked, with a round-capped check in the fill&apos;s contrast color, in a 48dp touch target with its label 11dp after the box. Checking takes 300ms: the box shrinks 1dp while the outline closes in, then the check grows from its corner; unchecking runs it back. Android draws no ghost checkbox, so it keeps the box. <code>pill</code> is kept for <code>@plexui/ui</code>'s API: the box is a circle on iOS and a square on Android either way, so the example leaves it out.</p>
  <p>Use <code>description</code> for a second line under the label, in the secondary color; VoiceOver and TalkBack read it after the label. By default the box is on the left of its label; <code>orientation="right"</code> places it after the label. Both act only next to a label, so they are offered only with one. Without a <code>label</code> the checkbox is a standalone control: pass <code>accessibilityLabel</code> to name it.</p>
  <p><code>disabled</code> dims the box and its label (on Android both keep their colors at half), keeps the checked and indeterminate states, and takes no presses.</p>
</>} code={overviewCode} previewMinHeight={168}>
  <RNExample demo="checkbox/Overview" />
</ComponentPreview>
