export const overviewCode = `import { List, ListItem, RadioGroup, ThemeProvider } from "@plexui/react-native"

const frequencies = [
  { value: "daily", label: "Daily", description: "A summary every morning." },
  { value: "weekly", label: "Weekly", description: "A digest every Monday." },
  { value: "monthly", label: "Monthly", description: "A recap on the first of the month." },
  { value: "never", label: "Never", description: "Only security alerts." },
]

const [frequency, setFrequency] = useState("daily")
const [description, setDescription] = useState(false)
const [disabled, setDisabled] = useState(false)
const [itemDisabled, setItemDisabled] = useState(false)
// The app's on color, set once on its ThemeProvider: "primary" (the default) or "accent"
const [selectionColor, setSelectionColor] = useState<"primary" | "accent">("primary")

<ThemeProvider selectionColor={selectionColor}>
  {/* Each option is a row; the chosen row ends with its mark */}
  <RadioGroup aria-label="Notification frequency" value={frequency} onChange={setFrequency} disabled={disabled}>
    <List header="Notification frequency">
      {frequencies.map((option) => (
        <ListItem
          key={option.value}
          label={option.label}
          description={description ? option.description : undefined}
          disabled={disabled || (itemDisabled && option.value === "monthly")}
          trailing={
            <RadioGroup.Item
              value={option.value}
              aria-label={option.label}
              disabled={itemDisabled && option.value === "monthly"}
            />
          }
        />
      ))}
    </List>
  </RadioGroup>
</ThemeProvider>`

### Overview

<ComponentPreview details={<>
  <p>On a phone, put the options in the rows of a <code>List</code> inside the <code>RadioGroup</code>: each <code>ListItem</code> names an option, and a <code>RadioGroup.Item</code> without children is its <code>trailing</code> mark, as in an iOS settings list or a section of a settings screen on Android. The list's header names the choice.</p>
  <p>Each option is a row of a <code>List</code>, and its <code>RadioGroup.Item</code> without children ends the row. On iOS the chosen row ends with the checkmark of a list row, in the text color (the accent, the iOS tint, when the <code>ThemeProvider</code>&apos;s <code>selectionColor</code> is the accent), in a 22pt column, and the others leave the column empty: iOS has no radio button. On Android each row ends with a radio button, a 20dp ring of 2dp, light gray when empty and the selection color (the <code>ThemeProvider</code>&apos;s <code>selectionColor</code>: the Plex primary unless set to the accent) with a 10dp dot when chosen, 23dp in from the card's edge, in a 48dp touch target. Choosing takes 200ms: the gray fills the ring inward, then shrinks to the dot as it takes the selection color. A disabled option keeps its colors at half. Give each item the row's label as its <code>aria-label</code>.</p>
  <p>A row's <code>description</code> puts a second line under its label, read after it by VoiceOver and TalkBack.</p>
  <p><code>disabled</code> on <code>RadioGroup</code> disables every option; on a single <code>RadioGroup.Item</code> it disables only that one, here Monthly, so that setting is offered only while the group is enabled. Set <code>disabled</code> on the row too, so its label dims with the option.</p>
  <p>A phone list is always a column and its mark always ends the row, so the example does not offer <code>direction</code> or <code>orientation</code>; a <code>RadioGroup.Item</code> with a label takes both outside a list.</p>
</>} code={overviewCode} previewMinHeight={168}>

</ComponentPreview>
