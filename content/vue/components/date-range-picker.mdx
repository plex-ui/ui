export const overviewCode = `<script setup lang="ts">
import { ref } from 'vue';
import { DateTime } from 'luxon';
import { DateRangePicker, type DateRange, type DateRangePickerProps, type DateRangeShortcut } from '@plexui/vue/components/DateRangePicker';
import { Field } from '@plexui/vue/components/Field';
import { FieldGroup } from '@plexui/vue/components/FieldGroup';
import { Select } from '@plexui/vue/components/Select';

const today = DateTime.local();
const minDate = today.minus({ days: 29 }).startOf('day');
const maxDate = today.endOf('day');
const dayShortcuts: DateRangeShortcut[] = [
  { label: 'Today', getDateRange: () => [today.startOf('day'), today.endOf('day')] },
  { label: 'Yesterday', getDateRange: () => [today.minus({ days: 1 }).startOf('day'), today.minus({ days: 1 }).endOf('day')] },
  { label: 'Last 7 days', getDateRange: () => [today.minus({ days: 6 }).startOf('day'), today.endOf('day')] },
  { label: 'Last 30 days', getDateRange: () => [today.minus({ days: 29 }).startOf('day'), today.endOf('day')] },
];
// Phone: the Period row offers the shortcuts and reads Custom for any other range. A shortcut the limits rule out
// is disabled, as in the calendar: whole days from min to max, and no more than maxRangeDays days.
const allows = ([start, end]: DateRange, min?: DateTime, max?: DateTime, maxRangeDays?: number) =>
  !(min && start < min.startOf('day')) &&
  !(max && end > max.endOf('day')) &&
  !(maxRangeDays != null && end.startOf('day') > start.startOf('day').plus({ days: maxRangeDays - 1 }));
const periodOptionsFor = (min?: DateTime, max?: DateTime, maxRangeDays?: number) =>
  dayShortcuts.map(({ label, getDateRange }) => ({ value: label, label, disabled: !allows(getDateRange(), min, max, maxRangeDays) }));
const periodOf = (range: DateRange | null) =>
  range
    ? dayShortcuts.find(({ getDateRange }) => {
        const [start, end] = getDateRange();
        return range[0].hasSame(start, 'day') && range[1].hasSame(end, 'day');
      })?.label ?? ''
    : '';

const value = ref<DateRange | null>(null);
const variant = ref<"outline" | "soft" | "ghost">("outline");
const size = ref<DateRangePickerProps["size"]>("lg"); // "3xs" to "3xl"
const pill = ref(true);
const block = ref(false);
const clearable = ref(true);
const triggerShowIcon = ref(true);
const dropdownIconType = ref<"dropdown" | "chevronDown" | "none">("dropdown");
const triggerDateFormat = ref("MM/dd/yy"); // or "MMM d", "MMM d, yyyy"
const triggerStepperUnit = ref("none"); // or "month"
const shortcuts = ref(false);
const limits = ref("none"); // or "past" (the last 30 days up to today)
const maxRangeDays = ref("none"); // or "7"
const weekStartsOn = ref("0"); // Sunday, or "1": Monday
const side = ref<"top" | "right" | "bottom" | "left">("bottom");
const sideOffset = ref(8);
const align = ref<"start" | "center" | "end">("center");
const alignOffset = ref(-5); // with a start or end alignment
const disabled = ref(false);
</script>

${overview}

<!-- Phone: the shortcuts are a pop-up in a row of their own -->

</template>`

## API

In the calendars, the arrow keys move focus by a day or a week, Home and End move it to the first or last day of the week row, and Page Up and Page Down move it by a month (January 31 goes to the last day of February). Focus moves across the visible months, and a day past them turns the calendars by a month first. Focus stays put when the day it would move to is disabled: outside `min` and `max`, or past `max-range-days` while a range is being picked. Enter and Space pick the focused day.

## Styling

Date Range Picker reads these component tokens. Override them at any scope to restyle it:

- `--input-border-color-invalid`: border color invalid (default `var(--color-border-danger-outline)`)
- `--input-outline-border-color`: outline border color (default `var(--color-border-primary-outline)`)
- `--input-outline-border-color-hover`: outline border color hover (default `light-dark(var(--alpha-25), var(--alpha-30))`)
- `--input-placeholder-text-color`: placeholder text color (default `var(--color-text-tertiary)`)
- `--input-text-color`: text color (default `var(--color-text)`)
- `--select-control-font-weight`: font weight (default `var(--font-weight-normal)`)
