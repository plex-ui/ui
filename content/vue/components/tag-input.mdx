import { VueDemo } from "@/components/docs/vue/VueDemo"

export const overview = `<div class="w-[480px]">
  <TagInput placeholder="Add a tag..." :size="size" :rows="Number(rows)" :pill="pill" />

export const suggestions = `<div class="flex w-full max-w-sm flex-col gap-3">
  <TagInput v-model:value="tags" placeholder="Add a skill..." :size="size" :pill="pill" />
  <div v-if="suggestions.some((s) => !tags.some((t) => t.value === s))" class="flex flex-wrap gap-1.5">
    <template v-for="s in suggestions" :key="s">
      <Button
        v-if="!tags.some((t) => t.value === s)"
        size="xs"
        color="secondary"
        variant="soft"
        :pill="pill"
        @click="tags = [...tags, { value: s, valid: true }]"
      >
        {{ s }}
      </Button>
    </template>
  </div>
</div>`

export const suggestionsCode = `<script setup lang="ts">
import { computed, ref } from "vue"
import { Button } from "@plexui/vue/components/Button"
import { TagInput, type Tag } from "@plexui/vue/components/TagInput"

const SUGGESTIONS = ["React", "TypeScript", "Next.js", "Tailwind CSS", "Node.js", "GraphQL", "PostgreSQL", "Redis", "Docker", "AWS"]

const tags = ref<Tag[]>([
  { value: "React", valid: true },
  { value: "TypeScript", valid: true },
])

const available = computed(() => {
  const selected = new Set(tags.value.map((t) => t.value))
  return SUGGESTIONS.filter((s) => !selected.has(s))
})
</script>

<template>
  <div class="flex flex-col gap-3">
    <TagInput v-model:value="tags" placeholder="Add a skill..." />
    <div v-if="available.length > 0" class="flex flex-wrap gap-1.5">
      <Button
        v-for="s in available"
        :key="s"
        size="xs"
        color="secondary"
        variant="soft"
        :pill="false"
        @click="tags = [...tags, { value: s, valid: true }]"
      >
        {{ s }}
      </Button>
    </div>
  </div>
</template>`
