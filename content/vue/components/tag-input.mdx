export const suggestionsCode = `<script setup lang="ts">
import { computed, ref } from "vue"
import { Button } from "@plexui/vue/components/Button"
import { TagInput, type Tag } from "@plexui/vue/components/TagInput"

const SUGGESTIONS = ["React", "TypeScript", "Next.js", "Tailwind CSS", "Node.js", "GraphQL", "PostgreSQL", "Redis", "Docker", "AWS"]

const tags = ref<Tag[]>([
  { value: "React", valid: true },
  { value: "TypeScript", valid: true },
])

const available = computed(() => {
  const selected = new Set(tags.value.map((t) => t.value))
  return SUGGESTIONS.filter((s) => !selected.has(s))
})
</script>

<template>
  <div class="flex flex-col gap-3">
    <TagInput v-model:value="tags" placeholder="Add a skill..." size="md" />
    <div v-if="available.length > 0" class="flex flex-wrap gap-1.5">
      <Button
        v-for="s in available"
        :key="s"
        size="xs"
        color="secondary"
        variant="soft"
        :pill="false"
        @click="tags = [...tags, { value: s, valid: true }]"
      >
        {{ s }}
      </Button>
    </div>
  </div>
</template>`

<UsageBlock>{`import { TagInput } from "@plexui/vue/components/TagInput";`}</UsageBlock>

On Phone, put a tag input under the label of a row in a `FieldGroup` (`Field row orientation="vertical"`) with `variant="ghost"`, as the [Form Examples](/vue/components/form-examples) put a textarea. The row is its frame and sets its insets and type, so the Phone previews show rows without the `size` and `pill` controls. To pick tags from a fixed list, use the tag row of the Form Examples, which opens a sheet.

## Styling

Tag Input reads these component tokens. Override them at any scope to restyle it:

- `--button-font-weight`: font weight (default `var(--font-weight-medium)`)
- `--input-outline-border-color`: outline border color (default `var(--color-border-primary-outline)`)
- `--input-outline-border-color-focus`: outline border color focus (default `var(--alpha-50)`)
- `--input-text-color`: text color (default `var(--color-text)`)
