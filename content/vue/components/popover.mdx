export const controllerCode = `<script setup lang="ts">
import { onBeforeUnmount, ref } from 'vue';
import { Button } from '@plexui/vue/components/Button';
import { Sparkles } from '@plexui/vue/components/Icon';
import { Popover, PopoverContent, PopoverTrigger } from '@plexui/vue/components/Popover';
import { Textarea } from '@plexui/vue/components/Textarea';
import { useBreakpoint } from '@plexui/vue/composables/useBreakpoints';

const wide = useBreakpoint('md');
const open = ref(false);
const prompt = ref('');
const submitting = ref(false);
let timer: ReturnType<typeof setTimeout> | undefined;
let session = 0;

onBeforeUnmount(() => {
  session++;
  clearTimeout(timer);
});

function setOpen(next: boolean) {
  session++;
  clearTimeout(timer);
  timer = undefined;
  if (next) {
    submitting.value = false;
    prompt.value = '';
  }
  open.value = next;
}

function setPrompt(value: string) {
  prompt.value = value;
}

function submitPrompt(close: () => void, shake: () => void) {
  if (!prompt.value) {
    shake();
    return;
  }
  if (timer !== undefined) return;
  submitting.value = true;
  const currentSession = session;
  timer = setTimeout(() => {
    if (session !== currentSession) return;
    timer = undefined;
    close();
  }, 2000);
}
</script>

<template>
${controller}
</template>`

### Nesting

All floating UI components in Plex UI can be infinitely nested by relying on `z-index: auto;` and natural stacking order.

Additionally, `esc` keypresses are handled independently as a stacking order.

## API

**Popover** (root)

<PropsTable data={[
  { prop: "open / v-model:open", type: "boolean", description: "Controlled open state; uncontrolled when omitted" },
  { prop: "@update:open", type: "(nextState: boolean) => void", description: "Emitted when the popover opens or closes" },
  { prop: "show-on-hover", type: "boolean", default: "false", description: "Open on hover and focus instead of click" },
  { prop: "hover-open-delay", type: "number", default: "150", description: "Delay before a hover popover opens (ms)" },
]} />

**PopoverTrigger**: Renders the element that toggles the popover. The trigger should be the user's point of interaction for revealing the popup. No specific props.

**PopoverAnchor**: Stands the popover at an element that is not its trigger, such as a bar button that keeps its own action; open the popover with `v-model:open`. A press on the element is not outside the popover, and focus returns to it on close. Prop: `element` (`HTMLElement | null`). `DialogContent`'s `confirm-close` uses it to ask at a phone Sheet's Close.

**PopoverContent**

<PropsTable data={[

  { prop: "max-width", type: "number", default: "700", description: "Maximum panel width in px, capped by the space available in the viewport" },
  { prop: "side", type: '"top" | "right" | "bottom" | "left"', default: '"bottom"', description: "Preferred placement; flips when it collides" },
  { prop: "align", type: '"start" | "center" | "end"', default: '"center"', description: "Alignment relative to trigger" },

  { prop: "align-offset", type: "number", default: "-5", description: "Offset from the start or end alignment (px)" },

  { prop: "translucent", type: "boolean", default: "false", description: "Translucent, blurred surface" },

  { prop: "class", type: "string", description: "Applied to the panel" },

**usePopoverController()**: Composable available in components inside `PopoverContent`. Returns `{ close: () => void, shake: () => void }`, the same functions the content's default slot receives.

## Styling

Popover reads these component tokens. Override them at any scope to restyle it:

- `--popover-radius`: radius (default `var(--radius-xl)`, `34px` at 767px and below, the corner of an iOS 27.2 popover)
- `--popover-mobile-background`: panel surface at 767px and below (default `light-dark(rgb(255 255 255 / 65%), var(--alpha-12))`: the glass of the phone menus, as iOS 27.2 draws a popover)
- `--popover-mobile-backdrop-filter`: blur behind the panel at 767px and below (default `--menu-mobile-backdrop-filter`, `blur(18px)`)
- `--popover-mobile-shadow`: shadow and edge of the panel at 767px and below (unset; the panel then takes `--menu-shadow`, the phone menus' raised shadow and edge)

A `translucent` panel keeps its own surface and shadow on phones.
