export const controller = `<Popover>
  <PopoverTrigger>
    <Button color="primary" :pill="false">Generate</Button>
  </PopoverTrigger>
  <PopoverContent side="right" v-slot="{ close, shake }">
    <form
      @submit.prevent="
        if (!prompt) return shake();
        submitting = true;
        submitPrompt().then(() => { submitting = false; prompt = ''; close(); });
      "
    >
      <textarea
        v-model="prompt"
        placeholder="Describe what you're using the model for, and we'll generate system instructions."

### Nesting

All floating UI components in Plex UI can be infinitely nested by relying on `z-index: auto;` and natural stacking order.

Additionally, `esc` keypresses are handled independently as a stacking order.

## Reference

**Popover** (root)

<PropsTable data={[
  { prop: "showOnHover", type: "boolean", description: "Open on hover and focus instead of click" },
  { prop: "open / v-model:open", type: "boolean", description: "Controlled open state; uncontrolled when omitted" },
]} />

**PopoverTrigger**: Renders the element that toggles the popover. The trigger should be the user's point of interaction for revealing the popup. No specific props.

**PopoverContent**

<PropsTable data={[
  { prop: "width", type: "number | string", description: "Panel width (px or CSS value)" },
  { prop: "minWidth", type: "number | string", default: "300", description: "Minimum panel width in px" },
  { prop: "maxWidth", type: "number | string", description: "Maximum panel width" },
  { prop: "side", type: '"top" | "right" | "bottom" | "left"', description: "Preferred placement" },

  { prop: "sideOffset", type: "number", description: "Offset from trigger (px)" },
  { prop: "alignOffset", type: "number", description: "Alignment offset (px)" },
]} />

**usePopoverController()**: Composable available in components inside `PopoverContent`. Returns `{ close: () => void, shake: () => void }`, the same functions the content's default slot receives.
