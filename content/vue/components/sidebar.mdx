  text

export const DOCS_LEVEL_VARS = ["a", "b", "c", "d"]

export const docsLevel = (list, depth, active, onSelect) => {
  const item = DOCS_LEVEL_VARS[depth]
  const leafButton = `<SidebarMenuSubButton :indent="${depth}" :is-active="${active}.id === ${item}.id" @click="${active} = ${item}${onSelect}">{{ ${item}.label }}</SidebarMenuSubButton>`
  if (depth === 3) {
    return `<SidebarMenuSubItem v-for="${item} in ${list}" :key="${item}.id">
  ${leafButton}
</SidebarMenuSubItem>`
  }
  return `<template v-for="${item} in ${list}" :key="${item}.id">
  <SidebarMenuItem v-if="${item}.items" :expanded="!!expanded[${item}.id]">
    <SidebarMenuSubButton :indent="${depth}" @click="expanded[${item}.id] = !expanded[${item}.id]">
      {{ ${item}.label }}<SidebarMenuChevron />
    </SidebarMenuSubButton>
    <SidebarMenuSub :open="!!expanded[${item}.id]">
${indent(docsLevel(`${item}.items`, depth + 1, active, onSelect), 6)}
    </SidebarMenuSub>
  </SidebarMenuItem>
  <SidebarMenuSubItem v-else>
    ${leafButton}
  </SidebarMenuSubItem>
</template>`
}

## Responsive composition

These application-local helpers choose the layout from the viewport. Desktop collapse remains independent of the mobile drawer. On a phone the drawer is a grouped page with a large title and a close button, as the sidebar of an iPhone app is: each menu is a card of 52px rows, a group label heads its section, a row that leads to a page ends with a disclosure chevron, and an item with a submenu keeps its own chevron, which turns down as its items open under it. The drawer itself is a modal dialog while it is open: focus moves into it and stays there, the page behind it is inert, Escape inside it closes it, and focus returns to the menu button. A swipe in from the edge it slides from opens it, and a swipe back closes it. Keep navigation selection handlers on the row buttons; popup and disclosure buttons do not close the drawer.

Save the helpers in `responsive-sidebar.ts` and the styles in `responsive-sidebar.module.css`:

```ts
import { defineComponent, h, type PropType } from 'vue';
import { NavigationButton } from '@plexui/vue/components/Button';
import { Page, PageBar, PageContent, PageHeader, PageLeading, PageTitle } from '@plexui/vue/components/Page';
import { Sidebar, SidebarInset, SidebarMobile, SidebarMobileMenuButton, SidebarProvider, useSidebar as sidebarContext, type SidebarCollapsible, type SidebarProviderVariant } from '@plexui/vue/components/Sidebar';
import { Toolbar, ToolbarTitle, ToolbarTrailing } from '@plexui/vue/components/Toolbar';
import { useBreakpoint as breakpoint } from '@plexui/vue/composables/useBreakpoints';
import s from './responsive-sidebar.module.css';
/** Leaf navigation closes the drawer; disclosure and popup buttons keep it open. */
export function isSidebarNavigationSelection(target: EventTarget | null) {
  if (!(target instanceof Element)) return false;
  const button = target.closest('[data-active]');
  return Boolean(button && !button.hasAttribute('aria-haspopup') &&
    !button.hasAttribute('aria-expanded') && !button.closest('li')?.querySelector(':scope > ul'));
}

export const ResponsiveSidebarProvider = defineComponent({
  name: 'ResponsiveSidebarProvider',
  inheritAttrs: false,
  props: {
    open: { type: Boolean, default: undefined },
    collapsible: { type: String as PropType<SidebarCollapsible>, default: 'offcanvas' },
    variant: { type: String as PropType<SidebarProviderVariant>, default: 'sidebar' },
  },
  setup(props, { attrs, slots }) {
    const wide = breakpoint('md');
    return () => h(SidebarProvider, {
      ...attrs,
      ...props,
      class: [s.Frame, attrs.class],
      open: wide.value ? props.open : true,
      variant: wide.value ? props.variant : 'sidebar',
    }, slots);
  },
});

export const ResponsiveSidebar = defineComponent({
  name: 'ResponsiveSidebar',
  inheritAttrs: false,
  setup(_, { attrs, slots }) {
    const sidebar = sidebarContext();
    return () => sidebar.isMobile
      ? h(SidebarMobile, { contained: true, class: s.Drawer, 'aria-label': 'Navigation' }, () =>
        h(Page, { surface: 'grouped' }, () => [
          h(Toolbar, { titleStyle: 'large' }, () => [
            h(ToolbarTitle, null, () => 'Navigation'),
            h(ToolbarTrailing, null, () => h(NavigationButton, { action: 'close', 'aria-label': 'Close menu', onClick: () => sidebar.setOpenMobile(false) })),
          ]),
          h(PageContent, { class: s.NavigationContent, onClick: (event: MouseEvent) => {
            if (!event.defaultPrevented && isSidebarNavigationSelection(event.target)) sidebar.setOpenMobile(false);
          } }, slots),
        ]))
      : h(Sidebar, attrs, slots);
  },
});

export const SidebarDemoPage = defineComponent({
  name: 'SidebarDemoPage',
  props: { title: { type: String, required: true } },
  setup(props, { slots }) {
    const sidebar = sidebarContext();
    return () => h(SidebarInset, { class: s.Inset }, () => sidebar.isMobile
      ? h(Page, null, () => [
        h(PageHeader, null, () => h(PageBar, null, () => [
          h(PageLeading, null, () => h(SidebarMobileMenuButton)),
          h(PageTitle, null, () => props.title),
        ])),
        h(PageContent, null, slots),
      ])
      : h('div', { class: 'p-6' }, [h('h1', { class: 'text-2xl font-semibold mb-4' }, props.title), slots.default?.()]));
  },
});
```

## API

### Structure

A Sidebar is composed of the following parts:

- `SidebarProvider`: Handles collapsible state and provides context to all child components.
- `SidebarLayout`: Wrapper for sidebar and main content.
- `Sidebar`: The sidebar container.
- `SidebarHeader` / `SidebarFooter`: Sticky at the top and bottom of the sidebar.
- `SidebarContent`: Scrollable content area.
- `SidebarGroup`: Section within the content, with optional label.
- `SidebarMenu` / `SidebarMenuItem` / `SidebarMenuButton`: Navigation menu items.
- `SidebarMenuSub` / `SidebarMenuSubButton`: Nested submenu items.
- `SidebarInset`: Main content area next to the sidebar.
- `SidebarTrigger`: Toggle button for collapsing the sidebar.

Use `Cmd+B` (Mac) or `Ctrl+B` (Windows/Linux) to toggle the sidebar with keyboard.

### SidebarProvider

Handles sidebar state and provides context to all child components. Wrap your application layout in this provider.

Composable to access sidebar state and controls from the `setup()` of any component within `SidebarProvider`. It returns a reactive object: read its fields (`sidebar.state`) rather than destructuring them outside a render.

- `openMobile` / `setOpenMobile`: Mobile menu open state
- `isMobile`: Whether the viewport is mobile
- `collapsible`: Current collapsible mode
- `toggleSidebar`: Toggle function for both desktop and mobile
