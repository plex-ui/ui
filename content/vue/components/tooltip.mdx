import { VueDemo } from "@/components/docs/vue/VueDemo"

export const base = `<Tooltip content="This is additional context that appears when the trigger is hovered or focused">
  <TooltipTriggerDecorator>Simple text with tooltip</TooltipTriggerDecorator>
</Tooltip>`

export const triggers = `<div class="flex items-center gap-4">
  <Tooltip content="Tooltip on button">
    <Button color="primary" size="lg">Sample button</Button>
  </Tooltip>
  <Tooltip content="Tooltip on span">
    <span>Inaccessible trigger</span>
  </Tooltip>
</div>`

export const conditional = `<Tooltip :content="disabled ? 'This action is disabled for reasons' : null">
  <Button color="primary" size="lg" :disabled="disabled">
    Sample button
  </Button>
</Tooltip>`

export const compact = `<Tooltip content="Copy to clipboard" compact>
  <Button color="secondary" variant="ghost" size="md"><Copy /></Button>
</Tooltip>`
