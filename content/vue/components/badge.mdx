  :color="color"
  :variant="variant"
  :size="size"
  :pill="pill"
  :class="leading === 'loading' ? 'gap-1.5' : undefined"
  :remove-label="'Remove ' + labels[leading]"

  <LoadingIndicator v-if="leading === 'loading'" />

</Badge>

<ComponentPreview language="vue" details={<>
  <p>Style a badge with <code>color</code> and <code>variant</code>, and set its <code>size</code>; <code>pill</code> rounds it fully and adds horizontal gutter. Its content can be composed: an icon before the text takes the badge's color, size and spacing, and a <code>LoadingIndicator</code> marks a task in progress, with a wider <code>gap-1.5</code> gap.</p>

To add the cross only at times, bind the handler as <code>:on-remove</code> and give it <code>undefined</code> otherwise, as here: Vue wraps an inline <code>@remove</code> expression in a function, so a listener written that way is always there. At <code>size="xl"</code> a badge has the height of a control, for a row of removable badges beside a button. Long text ends in an ellipsis and the cross stays.</p>

## Styling

Badge reads these component tokens. Override them at any scope to restyle it:

- `--badge-font-size-lg`: font size lg (default `var(--font-text-sm-size)`)
- `--badge-font-size-md`: font size md (default `var(--font-text-sm-size)`)
- `--badge-font-size-sm`: font size sm (default `var(--font-text-xs-size)`)
- `--badge-font-size-xl`: font size xl (default `var(--font-text-sm-size)`)
- `--badge-font-weight-lg`: font weight lg (default `var(--font-weight-semibold)`)
- `--badge-font-weight-md`: font weight md (default `var(--font-weight-semibold)`)
- `--badge-font-weight-sm`: font weight sm (default `var(--font-weight-semibold)`)
- `--badge-font-weight-xl`: font weight xl (default `var(--font-weight-semibold)`)
- `--badge-gutter-lg`: gutter lg (default `var(--control-gutter-xs)`)
- `--badge-gutter-md`: gutter md (default `var(--control-gutter-2xs)`)
- `--badge-gutter-sm`: gutter sm (default `calc(var(--control-gutter-2xs) - 1px)`)
- `--badge-gutter-xl`: gutter xl (default `var(--control-gutter-xs)`)
- `--badge-icon-font-size-lg`: icon font size lg (default `var(--font-text-md-size)`)
- `--badge-icon-font-size-md`: icon font size md (default `var(--font-text-md-size)`)
- `--badge-icon-font-size-sm`: icon font size sm (default `var(--font-text-xs-size)`)
- `--badge-icon-font-size-xl`: icon font size xl (default `var(--font-text-md-size)`)
- `--badge-indicator-size-lg`: indicator size lg (default `var(--font-text-sm-size)`)
- `--badge-indicator-size-md`: indicator size md (default `var(--font-text-xs-size)`)
- `--badge-indicator-size-sm`: indicator size sm (default `var(--font-text-xs-size)`)
- `--badge-indicator-size-xl`: indicator size xl (default `var(--font-text-sm-size)`)
- `--badge-radius-lg`: radius lg (default `var(--radius-sm)`)
- `--badge-radius-md`: radius md (default `var(--radius-xs)`)
- `--badge-radius-sm`: radius sm (default `var(--radius-xs)`)
- `--badge-radius-xl`: radius xl (default `var(--radius-md)`)
- `--badge-size-lg`: size lg (default `var(--control-size-2xs)`)
- `--badge-size-md`: size md (default `var(--control-size-3xs)`)
- `--badge-size-sm`: size sm (default `calc(var(--control-size-3xs) - 2px)`)
- `--badge-size-xl`: size xl (default `2rem`)
- `--badge-tracking-lg`: tracking lg (default `var(--tracking-normal)`)
- `--badge-tracking-md`: tracking md (default `var(--tracking-normal)`)
- `--badge-tracking-sm`: tracking sm (default `var(--tracking-wide)`)
- `--badge-tracking-xl`: tracking xl (default `var(--tracking-normal)`)
