export const overviewCode = `<script setup lang="ts">
import { ref } from 'vue';
import { FileUpload } from '@plexui/vue/components/FileUpload';

const files = ref<File[]>(initialFiles);
const errorMessage = ref(false);
const disabled = ref(false);
</script>

<template>
  <FileUpload
    v-model:value="files"
    accept=".pdf,.png,.jpg,.jpeg"
    multiple
    :max-files="5"
    :max-size="10 * 1024 * 1024"
    :error-message="errorMessage ? 'Please upload at least 2 documents.' : undefined"
    :disabled="disabled"
  />
</template>`

export const fileList = `<div ref="fileCards" class="w-[360px]">
  <FileCardGroup v-if="files.length > 0">
    <FileCard
      v-for="(file, index) in files"
      :key="file.name + '-' + file.size"
      :file="file"
      :on-remove="removable ? () => removeFileWithFocus($refs.fileCards, index, () => { files = files.filter((_, i) => i !== index) }) : undefined"
    />
  </FileCardGroup>
  <div role="status" tabindex="-1" data-file-removal-status :class="files.length > 0 ? 'sr-only' : 'text-center text-sm text-tertiary'">
    {{ files.length > 0 ? '' : 'Files removed' }}
  </div>
</div>`

export const fileListControls = [
  { name: "removable", label: "onRemove", type: "boolean", default: true },
]

## Styling

File Upload reads these component tokens. Override them at any scope to restyle it:

- `--field-error-margin-top`: error margin top (default `0.5rem`)
- `--file-upload-area-border-radius`: area border radius (default `var(--radius-xl)`)
- `--file-upload-area-min-height`: area min height (default `10rem`)
- `--file-upload-gutter`: gutter (default `var(--floating-input-gutter)`)
- `--file-upload-list-gap`: list gap (default `0.5rem`)
- `--floating-input-background`: background (default `var(--color-surface)`)
- `--floating-input-border-color`: border color
- `--floating-input-border-color-hover`: border color hover
- `--floating-input-border-color-invalid`: border color invalid (default `var(--color-border-danger-outline)`)
- `--floating-input-transition-duration`: transition duration (default `80ms`)
- `--input-text-color`: text color (default `var(--color-text)`)
