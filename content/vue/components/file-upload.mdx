import { VueFileUploadDemo } from "@/components/docs/vue/VueFileUploadDemos"

export const overview = `<div class="w-[360px]">
  <FileUpload
    v-model:value="files"
    accept=".pdf,.png,.jpg,.jpeg"
    multiple
    :max-files="5"
    :max-size="10 * 1024 * 1024"
  />
</div>`

export const withFiles = `<div class="w-[360px]">
  <FileUpload
    v-model:value="files"
    accept=".pdf,.png"
    multiple
    :max-files="5"
  />
</div>`

export const error = `<div class="w-[360px]">
  <FileUpload
    v-model:value="files"
    accept=".pdf"
    multiple
    error-message="Please upload at least 2 documents."
  />
</div>`

export const disabled = `<div class="w-[360px]">
  <FileUpload disabled />
</div>`

export const readOnly = `<div class="w-[360px]">
  <FileCardGroup>
    <FileCard :file="file" />
  </FileCardGroup>
</div>`

export const withRemove = `<div v-if="file" class="w-[360px]">
  <FileCardGroup>
    <FileCard :file="file" @remove="file = null" />
  </FileCardGroup>
</div>
<div v-else class="w-[360px] text-center text-sm text-tertiary">File removed</div>`

export const multipleFiles = `<div class="w-[360px]">
  <FileCardGroup>
    <FileCard
      v-for="(file, index) in files"
      :key="file.name + '-' + file.size"
      :file="file"
      @remove="files = files.filter((_, i) => i !== index)"
    />
  </FileCardGroup>
</div>`

export const readOnlyList = `<div class="w-[360px]">
  <FileCardGroup>
    <FileCard v-for="file in files" :key="file.name + '-' + file.size" :file="file" />
  </FileCardGroup>
</div>`
