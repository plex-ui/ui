export const controllerCode = `import { useEffect, useRef, useState, type FormEvent } from 'react';
import { Button } from '@plexui/ui/components/Button';
import { Sparkles } from '@plexui/ui/components/Icon';
import { Popover, usePopoverController } from '@plexui/ui/components/Popover';
import { Textarea } from '@plexui/ui/components/Textarea';
import { useBreakpoint } from '@plexui/ui/hooks/useBreakpoints';

function PopoverTextarea({
  value,
  onChange,
}: {
  value?: string;
  onChange?: (next: string) => void;
}) {
  return (
    <Textarea
      value={value}
      onChange={(evt) => onChange?.(evt.target.value)}
      aria-label="Prompt"
      variant="ghost"
      gutterSize="xl"
      rows={3}
      className="w-full"
      placeholder="Describe what you're using the model for, and we'll generate system instructions."
    />
  );
}

function PopoverActionBar({ loading }: { loading?: boolean }) {
  return (
    <div className="px-3 pb-3 flex items-center justify-between">
      <div className="flex gap-1.5 items-center font-normal text-sm text-tertiary">
        <Sparkles width={16} height={16} />
        Free beta
      </div>
      <Button color="primary" size="xs" pill={false} loading={loading} type="submit">
        Create
      </Button>
    </div>
  );
}

function PopoverControllerForm({ value, onChange, submitting, onSubmit }: {
  value: string;
  onChange: (value: string) => void;
  submitting: boolean;
  onSubmit: (close: () => void, shake: () => void) => void;
}) {
  const { shake, close } = usePopoverController();

  const handleSubmit = (evt: FormEvent) => {
    evt.preventDefault();
    onSubmit(close, shake);
  };

  return (
    <form className="pt-2" onSubmit={handleSubmit}>
      <PopoverTextarea value={value} onChange={onChange} />
      <PopoverActionBar loading={submitting} />
    </form>
  );
}

export function PopoverControllerDemo() {
  const wide = useBreakpoint('md');
  const [open, setOpen] = useState(false);
  const [value, setValue] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
  const session = useRef(0);

  useEffect(() => () => {
    session.current++;
    clearTimeout(timer.current);
  }, []);

  const changeOpen = (next: boolean) => {
    session.current++;
    clearTimeout(timer.current);
    timer.current = undefined;
    if (next) {
      setSubmitting(false);
      setValue('');
    }
    setOpen(next);
  };

  const submitPrompt = (close: () => void, shake: () => void) => {
    if (!value) {
      shake();
      return;
    }
    if (timer.current !== undefined) return;
    setSubmitting(true);
    const currentSession = session.current;
    timer.current = setTimeout(() => {
      if (session.current !== currentSession) return;
      timer.current = undefined;
      close();
    }, 2000);
  };

  return (
    <Popover open={open} onOpenChange={changeOpen}>
      <Popover.Trigger>
        <Button color="primary" pill={false}>Generate</Button>
      </Popover.Trigger>
      <Popover.Content side={wide ? 'right' : 'bottom'}>
        <PopoverControllerForm value={value} onChange={setValue} submitting={submitting} onSubmit={submitPrompt} />
      </Popover.Content>
    </Popover>
  );
}`

### Nesting

All floating UI components in Plex UI can be infinitely nested by relying on `z-index: auto;` and natural stacking order.

Additionally, `esc` keypresses are handled independently as a stacking order.

## API

**Popover** (root)

<PropsTable data={[
  { prop: "open", type: "boolean", description: "Controlled open state; uncontrolled when omitted" },
  { prop: "onOpenChange", type: "(nextState: boolean) => void", description: "Called when the popover opens or closes" },
  { prop: "showOnHover", type: "boolean", default: "false", description: "Open on hover and focus instead of click" },
  { prop: "hoverOpenDelay", type: "number", default: "150", description: "Delay before a hover popover opens (ms)" },
]} />

**Popover.Trigger**: Renders the element that toggles the popover. The trigger should be the user's point of interaction for revealing the popup. No specific props.

**Popover.Content**

<PropsTable data={[
  { prop: "width", type: 'number | "auto"', default: '"auto"', description: "Panel width in px" },
  { prop: "minWidth", type: 'number | "auto"', default: "300", description: "Minimum panel width in px" },
  { prop: "maxWidth", type: "number", default: "700", description: "Maximum panel width in px, capped by the space available in the viewport" },
  { prop: "side", type: '"top" | "right" | "bottom" | "left"', default: '"bottom"', description: "Preferred placement; flips when it collides" },
  { prop: "align", type: '"start" | "center" | "end"', default: '"center"', description: "Alignment relative to trigger" },
  { prop: "sideOffset", type: "number", default: "8", description: "Offset from trigger (px)" },
  { prop: "alignOffset", type: "number", default: "-5", description: "Offset from the start or end alignment (px)" },

**usePopoverController()**: Hook available inside `Popover.Content`. Returns `{ close: () => void, shake: () => void }`.

## Styling

Popover reads these component tokens. Override them at any scope to restyle it:

- `--popover-radius`: radius (default `var(--radius-xl)`, `34px` at 767px and below, the corner of an iOS 27.2 popover)
- `--popover-mobile-background`: panel surface at 767px and below (default `light-dark(rgb(255 255 255 / 65%), var(--alpha-12))`: the glass of the phone menus, as iOS 27.2 draws a popover)
- `--popover-mobile-backdrop-filter`: blur behind the panel at 767px and below (default `--menu-mobile-backdrop-filter`, `blur(18px)`)
- `--popover-mobile-shadow`: shadow and edge of the panel at 767px and below (unset; the panel then takes `--menu-shadow`, the phone menus' raised shadow and edge)

A `translucent` panel keeps its own surface and shadow on phones.
