## Tags in rows

<ComponentPreview details={<>With <code>fill</code> the control takes the rest of the line, so the tags of every row start at one edge. <code>action</code> holds the button at the end: it stays on the first line when the tags wrap, and on a phone it stands beside the label with the tags under it. <Field
  row
  fill
  label="Materials"
  action={
    <Select
      multiple
      block={false}
      align="end"
      aria-label="Add material"
      options={materials}
      value={value}
      onChange={(selected) => setValue(selected.map((option) => option.value))}
      trigger={({ onToggle }) => (
        <Tooltip content={wide ? "Add material" : null}>
          <Button uniform={wide} pill color="secondary" variant="soft" aria-label="Add material" onClick={onToggle}>
            {wide ? <Plus /> : "Add"}
          </Button>
        </Tooltip>
      )}
    />
  }
>
  <div className="flex flex-wrap gap-2">
    {chosen.map((option) => (
      <Badge
        key={option.value}
        size="xl"
        pill
        variant={option.color ? "solid" : "outline"}
        color={option.color ?? "secondary"}
        removeLabel={\`Remove \${option.label}\`}
        onRemove={() => remove(option.value)}
      >
        {option.label}
      </Badge>
    ))}
  </div>
</Field>`} previewMinHeight={200}>
