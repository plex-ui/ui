### File list

<ComponentPreview details={<>One file is a list of one. Turn off <code>onRemove</code> for read-only cards. After keyboard removal, focus moves to the next file, then the previous one, or the Files removed status when the list becomes empty, without scrolling. Pointer removal does not move focus if the remove button was not focused.</>} code={`${fileRemovalFocusCode}

function Example({ onRemove = true }) {
  const cards = useRef<HTMLDivElement>(null);
  const [files, setFiles] = useState<File[]>(initialFiles);

  return (
    <div ref={cards} className="w-[360px]">
      {files.length > 0 && <FileCardGroup>
        {files.map((file, index) => (
          <FileCard
            key={\`\${file.name}-\${file.size}\`}
            file={file}
            onRemove={onRemove ? () => removeFileWithFocus(cards.current, index, () => setFiles((prev) => prev.filter((_, i) => i !== index))) : undefined}
          />
        ))}
      </FileCardGroup>}
      <div role="status" tabIndex={-1} data-file-removal-status className={files.length > 0 ? "sr-only" : "text-center text-sm text-tertiary"}>
        {files.length > 0 ? "" : "Files removed"}
      </div>
    </div>
  );
}`}>
  <FileCardDemo />
</ComponentPreview>

## Styling

File Upload reads these component tokens. Override them at any scope to restyle it:

- `--field-error-margin-top`: error margin top (default `0.5rem`)
- `--file-upload-area-border-radius`: area border radius (default `var(--radius-xl)`)
- `--file-upload-area-min-height`: area min height (default `10rem`)
- `--file-upload-gutter`: gutter (default `var(--floating-input-gutter)`)
- `--file-upload-list-gap`: list gap (default `0.5rem`)
- `--floating-input-background`: background (default `var(--color-surface)`)
- `--floating-input-border-color`: border color
- `--floating-input-border-color-hover`: border color hover
- `--floating-input-border-color-invalid`: border color invalid (default `var(--color-border-danger-outline)`)
- `--floating-input-transition-duration`: transition duration (default `80ms`)
- `--input-text-color`: text color (default `var(--color-text)`)
