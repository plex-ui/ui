### Inline filter row

<ComponentPreview details={<>When <code>rows=1</code>, <code>TagInput</code> matches the height of an <code>Input</code> or <code>SelectControl</code> of the same size, so it composes cleanly into horizontal filter rows. Toggle <code>pill</code> on both controls to keep the pair visually consistent.</>} code={`const FIELD_OPTIONS = [
  { value: "email", label: "Email" },
  { value: "name", label: "Name" },
  { value: "role", label: "Role" },
  { value: "team", label: "Team" },
  { value: "country", label: "Country" },
]

const [field, setField] = useState("email")
const [tags, setTags] = useState<Tag[]>([
  { value: "ada@plexui.com", valid: true },
  { value: "grace@plexui.com", valid: true },
])
const fieldLabel = FIELD_OPTIONS.find((option) => option.value === field)?.label ?? ""

return (
  <div className="flex w-full max-w-2xl items-start gap-2">
    <Field label="Where" className="w-[160px] shrink-0">
      <Menu>
        <Menu.Trigger>
          <SelectControl size="md" pill={false} selected={!!field} className="w-full">
            {fieldLabel}
          </SelectControl>
        </Menu.Trigger>
        <Menu.Content minWidth={160} onCloseAutoFocus={(event) => event.preventDefault()}>
          {FIELD_OPTIONS.map((option) => (
            <Menu.CheckboxItem
              key={option.value}
              checked={field === option.value}
              onCheckedChange={() => setField(option.value)}
              indicatorVariant="ghost"
            >
              {option.label}
            </Menu.CheckboxItem>
          ))}
        </Menu.Content>
      </Menu>
    </Field>
    <Field label="Is in" className="flex-1">
      <TagInput size="md" pill={false} value={tags} onChange={setTags} placeholder="Add value, press Enter…" />
    </Field>
  </div>
)`} previewMinHeight={168}>
  <TagInputInlineFilterRowDemo />
</ComponentPreview>

### Suggestions

<ComponentPreview details={<>Combine <code>TagInput</code> with clickable buttons to let users add tags by clicking suggestions or typing manually. Suggestions disappear once selected.</>} code={`const SUGGESTIONS = ['React', 'TypeScript', 'Next.js', 'Tailwind CSS', 'Node.js', 'GraphQL', 'PostgreSQL', 'Redis', 'Docker', 'AWS'];

const [tags, setTags] = useState([
  { value: 'React', valid: true },
  { value: 'TypeScript', valid: true },
]);

const selectedValues = new Set(tags.map((t) => t.value));
const available = SUGGESTIONS.filter((s) => !selectedValues.has(s));

return (
  <div className="flex flex-col gap-3">
    <TagInput value={tags} onChange={setTags} placeholder="Add a skill..." size="md" />
    {available.length > 0 && (
      <div className="flex flex-wrap gap-1.5">
        {available.map((s) => (
          <Button key={s} size="xs" color="secondary" variant="soft" pill={false}
            onClick={() => setTags([...tags, { value: s, valid: true }])}>
            {s}
          </Button>
        ))}
      </div>
    )}
  </div>
);`} previewMinHeight={168}>
  <TagInputSuggestionsDemo />
</ComponentPreview>

## Styling

Tag Input reads these component tokens. Override them at any scope to restyle it:

- `--button-font-weight`: font weight (default `var(--font-weight-medium)`)
- `--input-outline-border-color`: outline border color (default `var(--color-border-primary-outline)`)
- `--input-outline-border-color-focus`: outline border color focus (default `var(--alpha-50)`)
- `--input-text-color`: text color (default `var(--color-text)`)
