export const overviewCode = `// An onboarding: a symbol, a title and a line on each page
const cards = [
  { icon: <Images />, title: "All your photos", text: "Every photo from your devices, in one library." },
  { icon: <Search />, title: "Find any photo", text: "Search by place, date or who is in the photo." },
  { icon: <Folders />, title: "Albums", text: "Keep a trip or an event together in an album." },
  // ...
]

const pagesId = useId()
const scroller = useRef<HTMLDivElement>(null)
const [page, setPage] = useState(0)
const pages = cards.slice(0, [[pages]])

<div
  id={pagesId}
  ref={scroller}
  className="flex snap-x snap-mandatory overflow-x-auto scroll-smooth motion-reduce:scroll-auto"
  onScroll={(event) => setPage(Math.round(event.currentTarget.scrollLeft / event.currentTarget.clientWidth))}
>
  {pages.map(({ icon, title, text }) => (
    <div key={title} className="flex w-full shrink-0 snap-center flex-col items-center gap-1 text-center">
      {icon}
      <div className="font-semibold">{title}</div>
      <div className="text-sm text-secondary">{text}</div>
    </div>
  ))}
</div>
<PageControl
  count={pages.length}
  value={page}
[[^background=automatic]]
  background="[[background]]"
[[/background=automatic]]
  aria-controls={pagesId}
  onChange={(index, { scrubbing }) => {
    setPage(index)
    scroller.current?.scrollTo({ left: index * scroller.current.clientWidth, behavior: scrubbing ? 'instant' : 'auto' })
  }}
/>`

The page control is a slider whose value is the page, read as Page 2 of 5, as VoiceOver reads the native control as adjustable. With the keyboard, Left and Right (mirrored in a right-to-left layout) or Up and Down move one page, Home and End go to the first and last page. The dots are presentation only.

## Styling

PageControl reads these component tokens. Override them at any scope to restyle it:

- `--page-control-height`, `--page-control-padding-inline`: the capsule (default `24px`, `12px`)
- `--page-control-gap`: space between dots (default `8px`)
- `--page-control-dot-size`, `--page-control-dot-size-small`, `--page-control-dot-size-smaller`: the full dot and the two shrinking ones (default `8px`, `6px`, `4px`)
- `--page-control-dot-color`, `--page-control-dot-color-current`: the dots (default `var(--alpha-30)`, `var(--color-text)`; the other dots in the tertiary label, `--mobile-label-tertiary`, on a phone with `touch.css`)
- `--page-control-background`: the capsule fill with `automatic` and `prominent` (default `var(--alpha-08)`)
- `--page-control-target-size`: height of the touch target (`44px` at 767px and below)
- `--page-control-transition-duration`: how fast dots change (default `200ms`, none with reduced motion)
