## DataTable

The examples import `users` (15 rows) and its `User` type from a local `./data` file. Header and cell renderers are functions that return JSX, rendered with TanStack Table's `flexRender`.

### Row selection

<ComponentPreview details={<>Set <code>enableRowSelection</code> to add a first column of checkboxes. The header checkbox selects every row on all pages and shows a mixed state when only some are selected. Bind <code>rowSelection</code> and <code>onRowSelectionChange</code> to read the selection; <code>getRowId</code> keys it by your own ids.</>} code={`import { useState } from "react"
import {
  DataTable,
  DataTableColumnHeader,
  statusCell,
  type ColumnDef,
  type RowSelectionState,
} from "@plexui/ui/components/DataTable"
import { users, type User } from "./data"

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: ({ column }) => <DataTableColumnHeader column={column} title="Name" />,
    cell: ({ row }) => <span className="font-medium">{row.original.name}</span>,
  },
  { accessorKey: "role", header: "Role" },
  { accessorKey: "status", header: "Status", cell: statusCell<User>((row) => row.status) },
]

export function UsersTable() {
  const [rowSelection, setRowSelection] = useState<RowSelectionState>({})

  return (
    <>
      <DataTable
        columns={columns}
        data={users}
        pageSize={5}
        enableRowSelection
        rowSelection={rowSelection}
        onRowSelectionChange={setRowSelection}
        getRowId={(user) => user.id}
      />
      <p className="mt-3 text-sm text-secondary">
        {Object.keys(rowSelection).length} of {users.length} rows selected
      </p>
    </>
  )
}`} previewMinHeight={340}>

## Styling

Table reads these component tokens. Override them at any scope to restyle it:

- `--table-border-color`: border color (default `var(--color-border)`)
- `--table-caption-color`: caption color (default `var(--color-text-tertiary)`)
- `--table-caption-font-size`: caption font size (default `var(--font-text-sm-size)`)
- `--table-cell-color`: cell color (default `var(--color-text)`)
- `--table-cell-font-size`: cell font size (default `var(--font-text-sm-size)`)
- `--table-cell-line-height`: cell line height (default `var(--font-text-sm-line-height)`)
- `--table-cell-padding`: cell padding (default `spacing(3) spacing(4)`)
- `--table-footer-background`: footer background (default `transparent`)
- `--table-footer-font-weight`: footer font weight (default `var(--font-weight-medium)`)
- `--table-header-background`: header background (default `transparent`)
- `--table-header-color`: header color (default `var(--color-text-secondary)`)
- `--table-header-font-weight`: header font weight (default `var(--font-weight-medium)`)
- `--table-row-hover-background`: row hover background (default `light-dark(var(--alpha-02), var(--alpha-04))`)
