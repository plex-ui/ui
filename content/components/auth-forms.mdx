Copy their styles and helpers with the example. Grouped is the default mobile surface; Plain is optional. Page uses base colors and Dialog uses elevated colors. Keep field values in the owner above the presentation wrapper so switching Page/Dialog preserves them. The error control demonstrates a visual state; it does not validate a code.

About you, login and verification default to `surface="grouped"`. Choose one surface for the steps of a flow; the palette adapts to light and dark mode on phones without changing the screen layout.

Auth screens use a shared `Page` shell in Phone view: a 54px toolbar area, a 10px gap before the form, and 16px side gutters. The `dialog` control switches to a mobile Sheet with its own toolbar. Individual input examples keep their component-only layout.

function VerificationExample() {

  const { title, description, resendLabel } = copy[channel];
  const presentation = useScreenPresentation();
  const wide = useBreakpoint('md');
  const formId = useId();
  const [attempt] = useState(createVerificationAttempt);
  const [error, setError] = useState(false);
  const [cooldown, setCooldown] = useState(false);
  const [seconds, setSeconds] = useState(RESEND_WAIT_SECONDS);
  const [verifying, setVerifying] = useState(false);
  const [mode, setMode] = useState<InputMode>('per-digit');
  useDemoControlSync('input', mode, setMode);
  const [otpValue, setOtpValue] = useState('');
  const [code, setCode] = useState('');
  const ready = isVerificationCodeReady(mode, mode === 'per-digit' ? otpValue : code);
  const updateVerifying = (value: boolean) => {
    attempt.setPending(value);
    setVerifying(value);
  };
  const updateError = (value: boolean) => {
    setError(value);
    if (value) updateVerifying(false);
  };
  const editCode = (value: string) => {
    attempt.edit(value);
    setError(false);
  };
  const completeCode = (value: string) => {
    if (!attempt.start('per-digit', value)) return;
    setError(false);
    setVerifying(true);
  };

  useEffect(() => {
    if (!cooldown) return;
    return startResendCountdown(setSeconds, () => setCooldown(false));

    <div className="w-full" data-verification-actions>
      <Button color="primary" size="3xl" type="submit" form={formId} pill block className="md:h-[3.25rem]"

            <SegmentedControl.Option value="text-field">Text field</SegmentedControl.Option>
          </SegmentedControl>
        </DemoControlRow>
        <DemoControlBoolean name="error" value={error} onChange={updateError} />
        <DemoControlBoolean name="cooldown" value={cooldown} onChange={setCooldown} />
        <DemoControlBoolean name="verifying" value={verifying} onChange={updateVerifying} />
        <ScreenPresentationControls {...presentation} />
      </div>
      <AuthFormStage keyboard="numeric" title={title} dialog={presentation.dialog}
        surface={presentation.surface} onDialogChange={presentation.setDialog} footer={actions}>
        <div className="w-[360px]">
          <div className="mb-6 text-center">
            <h3 data-auth-title className="text-2xl font-semibold tracking-tight">{title}</h3>
            <p data-auth-description className="text-secondary text-sm mt-1">{description}</p>
          </div>
          <form id={formId} className="flex flex-col gap-6 [--field-group-gap:13px]" noValidate aria-busy={verifying}
            onSubmit={(event) => {
              event.preventDefault();
              if (mode === 'text-field' && attempt.start(mode, code)) {
                setError(false);
                setVerifying(true);
              }
            }}>
            <div className="flex flex-col gap-2" data-verification-code>
              {mode === 'per-digit' ? (
                <div className="flex justify-center">
                  <OTPInput value={otpValue} onChange={(value) => { editCode(value); setOtpValue(value); }}
                    onComplete={completeCode} invalid={error}
                    errorMessage={error ? 'Incorrect code' : undefined} disabled={verifying}
                    className={authFieldsStyles.Otp} />
                </div>
              ) : (
                <AuthInput label="Code" value={code} invalid={error}
                  errorMessage={error ? 'Incorrect code' : undefined} disabled={verifying}
                  onChange={(event) => { editCode(event.target.value); setCode(event.target.value); }} inputMode="numeric" />
              )}
              <div data-verification-resend className="flex h-11 items-center justify-center">
              {mode === 'per-digit' && verifying ? (
                <LoadingIndicator role="status" aria-label="Checking code" />
              ) : (
              <Button type="button" color="secondary" variant="ghost" size="xl"
                className="h-11 w-full font-normal" disabled={cooldown || verifying}
                onClick={() => {
                  if (cooldown || verifying || attempt.isPending()) return;
                  attempt.reset();
                  setError(false);
                  setCooldown(true);
                }}>
                {cooldown ? \`\${resendLabel} (\${seconds}s)\` : resendLabel}
              </Button>
              )}
              </div>
            </div>
            {wide && actions}
          </form>
        </div>
      </AuthFormStage>
    </>
  );
}

import { AuthFormStage } from './AuthFormStage';

import { controlsTableStyle } from './ButtonDemos';

import { AuthFields } from './AuthFields';

export const PASSWORD_HINT = 'Your password must be at least 8 characters, include a number, an uppercase letter and a lowercase letter.';

type PasswordValidationErrors = { password?: string; confirmation?: string };

export function passwordErrors(password: string, confirmation: string): PasswordValidationErrors {
  return {
    password: !password ? 'Enter a password.' : !/^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d).{8,}$/.test(password) ? 'Password does not meet the requirements.' : undefined,
    confirmation: !confirmation ? 'Confirm your password.' : confirmation !== password ? 'Passwords do not match.' : undefined,
  };
}

export function CreatePasswordDemo() {
  const presentation = useScreenPresentation();
  const wide = useBreakpoint('md');
  const formId = useId();
  const [password, setPassword] = useState('');
  const [confirmation, setConfirmation] = useState('');
  const [submitted, setSubmitted] = useState(false);
  const [completed, setCompleted] = useState(false);
  const errors: PasswordValidationErrors = submitted ? passwordErrors(password, confirmation) : {};
  const submit = (event: React.FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    if (completed) return;
    setSubmitted(true);
    const invalid = passwordErrors(password, confirmation);
    if (invalid.password || invalid.confirmation) return;
    setCompleted(true);
    setSubmitted(false);
    setPassword('');
    setConfirmation('');
  };
  const action = completed ? (
    <Button color="primary" size="3xl" pill block onClick={() => setCompleted(false)}>Start over</Button>
  ) : (
    <Button color="primary" size="3xl" type="submit" form={formId} pill block>Continue</Button>
  );

  return (
    <>
      <div data-demo-controls style={controlsTableStyle}>
        <ScreenPresentationControls {...presentation} />
      </div>
      <AuthFormStage title="Create Password" surface={presentation.surface} dialog={presentation.dialog} onDialogChange={presentation.setDialog} footer={action}>
        <div className="w-full min-w-0">
          <div className="mb-6 text-center">
            <h3 data-auth-title className="text-2xl font-semibold tracking-tight">Create Password</h3>
          </div>
          <form id={formId} className="flex flex-col gap-6 [--field-group-gap:13px]" noValidate onSubmit={submit}>
            <AuthFields className="flex flex-col gap-2" footer={<span id={\`\${formId}-password-hint\`}>{PASSWORD_HINT}</span>}>
              <AuthInput label="Password" type="password"
                value={password} autoComplete="off" allowAutofillExtensions={false} disabled={completed}
                aria-describedby={!wide ? \`\${formId}-password-hint\` : undefined}
                onChange={(event) => { setPassword(event.target.value); setSubmitted(false); }}
                invalid={!!errors.password} errorMessage={errors.password} />
              <AuthInput label="Confirm Password" type="password"
                value={confirmation} autoComplete="off" allowAutofillExtensions={false} disabled={completed}
                onChange={(event) => { setConfirmation(event.target.value); setSubmitted(false); }}
                invalid={!!errors.confirmation} errorMessage={errors.confirmation} description={wide ? PASSWORD_HINT : undefined}
                aria-describedby={!wide ? \`\${formId}-password-hint\` : undefined} />
            </AuthFields>
            {completed && <div role="status"><Alert color="success" variant="soft" title="Preview complete"

}`} previewMinHeight={620}>
