### Multiple selection

<ComponentPreview details={<>Pair <code>FloatingLabelSelect</code> with a <code>multiple</code> <code>Select</code> to collect several values while keeping the floating-label treatment. The trigger shows the selected count and exposes a clear button that resets the whole list.</>} code={`import { FloatingLabelSelect } from "@plexui/ui/components/FloatingLabelSelect"
import { Select } from "@plexui/ui/components/Select"

const [skills, setSkills] = useState([])
const hasSelection = skills.length > 0
const display = skills.length === 1
  ? SKILLS.find((s) => s.value === skills[0])?.label
  : skills.length > 1 ? \`\${skills.length} skills\` : ""

<Select
  multiple
  clearable
  value={skills}
  options={SKILLS}
  onChange={(options) => setSkills(options.map((o) => o.value))}
  block
  trigger={({ open }) => (
    <FloatingLabelSelect
      label="Skills"
      selected={hasSelection}
      open={open}
      onClearClick={hasSelection ? () => setSkills([]) : undefined}
    >
      {display}
    </FloatingLabelSelect>
  )}
/>`} previewMinHeight={168}>
  <FloatingLabelSelectMultiDemo />
</ComponentPreview>

### Country select

<ComponentPreview details={<>Use <code>FloatingLabelSelect</code> alongside <code>FloatingLabelInput</code> for consistent form styling. The select trigger shares the same floating-label visual shell — the label floats up when a value is selected.</>} code={`import { FloatingLabelSelect } from "@plexui/ui/components/FloatingLabelSelect"
import { Menu } from "@plexui/ui/components/Menu"

const [country, setCountry] = useState(null)
const selected = COUNTRIES.find((c) => c.value === country)

<Menu>
  <Menu.Trigger>
    <FloatingLabelSelect
      label="Country"
      selected={!!selected}
      onClearClick={() => setCountry(null)}
    >
      {selected?.flag} {selected?.label}
    </FloatingLabelSelect>
  </Menu.Trigger>
  <Menu.Content>
    <Menu.RadioGroup value={country} onChange={setCountry}>
      {COUNTRIES.map((c) => (
        <Menu.RadioItem key={c.value} value={c.value}>
          {c.flag} {c.label}
        </Menu.RadioItem>
      ))}
    </Menu.RadioGroup>
  </Menu.Content>
</Menu>`} previewMinHeight={168}>
  <FloatingLabelSelectCountryDemo />
</ComponentPreview>
