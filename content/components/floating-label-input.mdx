### Select

### Multiple selection

<ComponentPreview details={<>Pair <code>FloatingLabelSelect</code> with a <code>multiple</code> <code>Select</code> to collect several values while keeping the floating-label treatment. The trigger shows the selected count and exposes a clear button that resets the whole list.</>} code={`import { FloatingLabelSelect } from "@plexui/ui/components/FloatingLabelSelect"
import { Select } from "@plexui/ui/components/Select"

const [skills, setSkills] = useState<string[]>([])
const hasSelection = skills.length > 0
const display =
  skills.length === 1
    ? (skillOptions.find((option) => option.value === skills[0])?.label ?? "")
    : skills.length > 1
      ? \`\${skills.length} skills\`
      : ""
const trigger = useCallback(
  ({ open }: { open: boolean }) => (
    <FloatingLabelSelect
      label="Skills"
      selected={hasSelection}
      open={open}
      onClearClick={hasSelection ? () => setSkills([]) : undefined}
    >
      {display}
    </FloatingLabelSelect>
  ),
  [hasSelection, display],
)

<div className="w-[360px]">
  <Select
    multiple
    clearable
    value={skills}
    options={skillOptions}
    onChange={(options) => setSkills(options.map((option) => option.value))}
    searchPlaceholder="Search skills..."
    listMinWidth={360}
    block
    trigger={trigger}
    checkPosition="end"
  />
</div>`} previewMinHeight={168}>
  <FloatingLabelSelectMultiDemo />
</ComponentPreview>

### Country select

<ComponentPreview details={<>Use <code>FloatingLabelSelect</code> alongside <code>FloatingLabelInput</code> for consistent form styling. The select trigger shares the same floating-label visual shell. The label floats up when a value is selected. <code>CountryLabel</code> is a helper of the example that shows a flag next to the name.</>} code={`import { FloatingLabelSelect } from "@plexui/ui/components/FloatingLabelSelect"
import { Select, type Option } from "@plexui/ui/components/Select"

// Outside the component: Select needs a stable OptionView
const CountryOptionView = ({ value, label }: Option) => <CountryLabel code={value} label={label} />

const [country, setCountry] = useState("")
const selected = countries.find((option) => option.value === country)
const trigger = useCallback(
  ({ open }: { open: boolean }) => (
    <FloatingLabelSelect
      label="Country"
      selected={!!selected}
      open={open}
      onClearClick={selected ? () => setCountry("") : undefined}
    >
      {selected && <CountryLabel code={selected.value} label={selected.label} />}
    </FloatingLabelSelect>
  ),
  [selected],
)

<div className="w-[360px]">
  <Select
    value={country}
    options={countries}
    onChange={(option) => setCountry(option.value)}
    searchPlaceholder="Search countries..."
    OptionView={CountryOptionView}
    listMinWidth={360}
    block
    trigger={trigger}
    checkPosition="end"
  />
</div>`} previewMinHeight={168}>
  <FloatingLabelSelectCountryDemo />
</ComponentPreview>

## Styling

Float Input reads these component tokens. Override them at any scope to restyle it:

- `--field-error-color`: error color (default `var(--color-text-danger-outline)`)
- `--field-error-font-size`: error font size (default `0.75rem`)
- `--field-error-line-height`: error line height (default `var(--font-text-xs-line-height)`)
- `--field-error-margin-top`: error margin top (default `0.5rem`)
- `--floating-input-background`: background (default `var(--color-surface)`)
- `--floating-input-border-color`: border color
- `--floating-input-border-color-focus`: border color focus (default `var(--alpha-50)`)
- `--floating-input-border-color-hover`: border color hover
- `--floating-input-border-color-invalid`: border color invalid (default `var(--color-border-danger-outline)`)
- `--floating-input-border-radius`: border radius (default `var(--radius-full)`)
- `--floating-input-gutter`: gutter (default `1.25rem`)
- `--floating-input-height`: height (default `3.25rem`)
- `--floating-input-label-color`: label color (default `var(--color-text-tertiary)`)
- `--floating-input-label-color-focus`: label color focus (default `var(--color-text)`)
- `--floating-input-label-color-invalid`: label color invalid (default `var(--color-text-danger-outline)`)
- `--floating-input-label-scale`: label scale (default `0.88`)
- `--floating-input-transition-duration`: transition duration (default `80ms`)
