For plain markup tables, use [Table](/components/table): Data Table draws its rows with its parts.

The examples import `users` (15 rows) and its `User` type from a local `./data` file. Header and cell renderers are functions that return JSX, rendered with TanStack Table's `flexRender`.

### Overview

<ComponentPreview details={<>Use <code>variant="bordered"</code> for a card-like table with a rounded border and tinted header background. Toggle <code>empty</code> to supply no rows without resetting sorting or changing columns. <code>emptyMessage</code> selects the default message, custom text, or rich content. Filtering and its no-results state are shown separately.</>} code={`import { useState } from "react"
import { DataTable, DataTableColumnHeader, statusCell, type ColumnDef } from "@plexui/ui/components/DataTable"
import { users, type User } from "./data"

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: ({ column }) => <DataTableColumnHeader column={column} title="Name" />,
    cell: ({ row }) => <span className="font-medium">{row.original.name}</span>,
  },
  { accessorKey: "role", header: "Role" },
  { accessorKey: "department", header: "Department" },
  { accessorKey: "status", header: "Status", cell: statusCell<User>((row) => row.status) },
]

const previewUsers = users.slice(0, 6)
const emptyUsers: User[] = []
const emptyMessages = {
  default: undefined,
  text: "No users found.",
  rich: <strong>No users found.</strong>,
}

export function UsersTable() {
  const [empty, setEmpty] = useState(false)
  const [emptyMessage, setEmptyMessage] = useState<keyof typeof emptyMessages>("text")

  return <DataTable columns={columns} data={empty ? emptyUsers : previewUsers}
    variant="bordered" pageSize={Infinity} emptyMessage={emptyMessages[emptyMessage]} />
}`} previewMinHeight={300}>

### Row click

<ComponentPreview details={<>Pass <code>onRowClick</code> to make rows clickable. Clickable rows also take Tab and call it on Enter or Space.</>} code={`import { useState } from "react"
import { DataTable, DataTableColumnHeader, type ColumnDef } from "@plexui/ui/components/DataTable"
import { users, type User } from "./data"

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: ({ column }) => <DataTableColumnHeader column={column} title="Name" />,
    cell: ({ row }) => <span className="font-medium">{row.original.name}</span>,
  },
  { accessorKey: "email", header: "Email" },
  { accessorKey: "role", header: "Role" },
]

export function UsersTable() {
  const [selectedUser, setSelectedUser] = useState<User | null>(null)

  return (
    <>
      <DataTable columns={columns} data={users.slice(0, 8)} pageSize={4} onRowClick={setSelectedUser} />
      <p className="mt-3 text-sm text-secondary">
        {selectedUser
          ? "Clicked: " + selectedUser.name + " (" + selectedUser.id + ") - " + selectedUser.department
          : "Click a row to see selected user details."}
      </p>
    </>
  )
}`} previewMinHeight={300}>

### Row selection

<ComponentPreview details={<>Set <code>enableRowSelection</code> to add a first column of checkboxes. The header checkbox selects every row on all pages and shows a mixed state when only some are selected. Bind <code>rowSelection</code> and <code>onRowSelectionChange</code> to read the selection; <code>getRowId</code> keys it by your own ids.</>} code={`import { useState } from "react"
import {
  DataTable,
  DataTableColumnHeader,
  statusCell,
  type ColumnDef,
  type RowSelectionState,
} from "@plexui/ui/components/DataTable"
import { users, type User } from "./data"

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: ({ column }) => <DataTableColumnHeader column={column} title="Name" />,
    cell: ({ row }) => <span className="font-medium">{row.original.name}</span>,
  },
  { accessorKey: "role", header: "Role" },
  { accessorKey: "status", header: "Status", cell: statusCell<User>((row) => row.status) },
]

export function UsersTable() {
  const [rowSelection, setRowSelection] = useState<RowSelectionState>({})

  return (
    <>
      <DataTable
        columns={columns}
        data={users}
        pageSize={5}
        enableRowSelection
        rowSelection={rowSelection}
        onRowSelectionChange={setRowSelection}
        getRowId={(user) => user.id}
      />
      <p className="mt-3 text-sm text-secondary">
        {Object.keys(rowSelection).length} of {users.length} rows selected
      </p>
    </>
  )
}`} previewMinHeight={340}>
  <TableRowSelectionDemo />
</ComponentPreview>

### Cell variants

<ComponentPreview details={<>Column helpers render common cells: <code>idCell</code> (monospace ID with copy-on-click), <code>statusCell</code> (status badge), <code>dateTimeCell</code> (formatted date) and <code>tooltipHeader</code> (header with an info tooltip).     cell: ({ row }) => <span className="font-medium">{money.format(row.original.salary)}</span>,

### Filtering

<ComponentPreview details={<>Control search with <code>globalFilter</code> and <code>onGlobalFilterChange</code>. The filter searches across all columns.</>} code={`import { useState } from "react"
import { Input } from "@plexui/ui/components/Input"
import { DataTable, DataTableColumnHeader, statusCell, type ColumnDef } from "@plexui/ui/components/DataTable"
import { users, type User } from "./data"

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: ({ column }) => <DataTableColumnHeader column={column} title="Name" />,
    cell: ({ row }) => <span className="font-medium">{row.original.name}</span>,
  },
  { accessorKey: "role", header: "Role" },
  { accessorKey: "department", header: "Department" },
  { accessorKey: "status", header: "Status", cell: statusCell<User>((row) => row.status) },
]

export function UsersTable() {
  const [globalFilter, setGlobalFilter] = useState("")

  return (
    <>
      <Input
        value={globalFilter}
        onChange={(event) => setGlobalFilter(event.target.value)}
        placeholder="Search name, role, department..."
      />
      <DataTable
        columns={columns}
        data={users}
        globalFilter={globalFilter}
        onGlobalFilterChange={setGlobalFilter}
        pageSize={5}
      />
    </>
  )
}`} previewMinHeight={340}>
